## Middleware Check Pattern Implementation

### 🛠️ Complete Implementation for ShopPulse

#### **Architecture Overview**

```
┌─────────────────────────────────────────────────────────┐
│       ShopPulse Middleware-Based Access Control          │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  Edge/CDN Layer (Optional - Cloudflare/Vercel):         │
│  └── middleware.ts executes FIRST                       │
│      ├── Runs before API handlers                       │
│      ├── Can access Edge KV store (ultra-fast)          │
│      └── Blocks request before reaching app server      │
│                                                          │
│  Request Flow:                                           │
│  1. GET /api/store_123/products                         │
│                                                          │
│  2. Middleware intercepts:                              │
│     - Extract storeId from URL                          │
│     - Check Edge KV cache (if available)                │
│     - Or check in-memory cache (5 min TTL)              │
│                                                          │
│  3. If cache miss:                                       │
│     - Query database (store + user + subscription)      │
│     - Update cache                                      │
│                                                          │
│  4. Validate subscription status:                       │
│     - ACTIVE/PAST_DUE → Allow                          │
│     - EXPIRED/CANCELED → Block (402)                   │
│                                                          │
│  5. If allowed:                                          │
│     - Add headers (x-user-id, x-subscription-status)    │
│     - Continue to API handler                           │
│                                                          │
│  6. If blocked:                                          │
│     - Return 402 error immediately                      │
│     - Never reaches API handler                         │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

**Step 1: Enhanced Middleware with Caching**

```typescript
// File: src/middleware.ts

import { clerkMiddleware } from '@clerk/nextjs/server';
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

// In-memory cache for subscription status
interface CacheEntry {
  status: 'ACTIVE' | 'EXPIRED' | 'PAST_DUE';
  userId: string;
  timestamp: number;
}

const subscriptionCache = new Map<string, CacheEntry>();
const CACHE_TTL = 5 * 60 * 1000; // 5 minutes

export default clerkMiddleware(async (auth, req: NextRequest) => {
  // Skip middleware for non-API routes
  if (!req.nextUrl.pathname.startsWith('/api/')) {
    return NextResponse.next();
  }
  
  // Extract storeId from URL pattern: /api/[storeId]/...
  const pathParts = req.nextUrl.pathname.split('/');
  if (pathParts.length < 3) {
    return NextResponse.next();
  }
  
  const storeId = pathParts[2];
  
  // Skip check for certain routes
  const skipRoutes = ['/api/webhook', '/api/auth', '/api/health'];
  if (skipRoutes.some(route => req.nextUrl.pathname.startsWith(route))) {
    return NextResponse.next();
  }
  
  // For POST/PATCH/DELETE, require Clerk authentication
  if (['POST', 'PATCH', 'DELETE', 'PUT'].includes(req.method)) {
    await auth.protect();
    return NextResponse.next();
  }
  
  // For GET requests, check subscription status
  if (req.method === 'GET') {
    const accessCheck = await checkStoreSubscriptionInMiddleware(storeId);
    
    if (!accessCheck.allowed) {
      return NextResponse.json(
        {
          error: 'Store unavailable',
          code: 'SUBSCRIPTION_EXPIRED',
          message: accessCheck.message,
          status: accessCheck.status
        },
        { 
          status: 402,
          headers: {
            'X-Store-Status': accessCheck.status,
            'Cache-Control': 'no-store' // Don't cache error responses
          }
        }
      );
    }
    
    // Add subscription info to headers for API handlers
    const response = NextResponse.next();
    response.headers.set('X-User-Id', accessCheck.userId);
    response.headers.set('X-Subscription-Status', accessCheck.status);
    
    // Add grace period warning if applicable
    if (accessCheck.status === 'PAST_DUE') {
      response.headers.set('X-Grace-Period', 'true');
    }
    
    return response;
  }
  
  return NextResponse.next();
});

/**
 * Check store subscription status (middleware-optimized version)
 * Uses in-memory cache to minimize database queries
 */
async function checkStoreSubscriptionInMiddleware(storeId: string) {
  // Check cache first
  const cached = subscriptionCache.get(storeId);
  if (cached && Date.now() - cached.timestamp < CACHE_TTL) {
    return createAccessResult(cached.status, cached.userId);
  }
  
  try {
    // Import prisma client (lazy-loaded for middleware)
    const prismadb = (await import('@/lib/prismadb')).default;
    
    // Query database
    const store = await prismadb.store.findUnique({
      where: { id: storeId },
      select: {
        id: true,
        userId: true,
        user: {
          select: {
            id: true,
            isActive: true,
            subscription: {
              select: {
                status: true
              }
            }
          }
        }
      }
    });
    
    if (!store) {
      return {
        allowed: false,
        status: 'NOT_FOUND',
        message: 'Store not found',
        userId: null
      };
    }
    
    const subscription = store.user.subscription;
    const subscriptionStatus = subscription?.status || 'NONE';
    
    // Update cache
    subscriptionCache.set(storeId, {
      status: subscriptionStatus as any,
      userId: store.userId,
      timestamp: Date.now()
    });
    
    return createAccessResult(subscriptionStatus, store.userId);
    
  } catch (error) {
    console.error('[MIDDLEWARE] Error checking subscription:', error);
    
    // Fail closed: deny access on errors
    return {
      allowed: false,
      status: 'ERROR',
      message: 'Unable to verify store status',
      userId: null
    };
  }
}

function createAccessResult(status: string, userId: string) {
  const allowedStatuses = ['ACTIVE', 'COMPLETING', 'PAST_DUE'];
  
  return {
    allowed: allowedStatuses.includes(status),
    status,
    message: allowedStatuses.includes(status)
      ? 'Store accessible'
      : 'Store unavailable due to subscription status',
    userId
  };
}

// Export cache invalidation function
export function invalidateMiddlewareCache(storeId: string) {
  subscriptionCache.delete(storeId);
}

export const config = {
  matcher: [
    '/((?!_next|[^?]*\\.(?:html?|css|js(?!on)|jpe?g|webp|png|gif|svg|ttf|woff2?|ico|csv|docx?|xlsx?|zip|webmanifest)).*)',
    '/(api|trpc)(.*)',
  ],
};
```

**Step 2: Edge KV Integration (For Vercel/Cloudflare)**

```typescript
// File: src/middleware-edge.ts (Vercel Edge Runtime)

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { get, set } from '@vercel/edge-config';

export const config = {
  runtime: 'edge', // Run at edge globally
  matcher: '/api/:storeId*'
};

export async function middleware(req: NextRequest) {
  const storeId = req.nextUrl.pathname.split('/')[2];
  
  // Check Edge Config (ultra-fast, replicated globally)
  const cacheKey = `store:${storeId}:subscription`;
  let subscriptionStatus = await get(cacheKey);
  
  if (!subscriptionStatus) {
    // Cache miss - fetch from origin
    // Note: Can't query database directly in edge runtime
    // Must call an API endpoint on your origin server
    const response = await fetch(
      `${req.nextUrl.origin}/api/internal/check-subscription/${storeId}`,
      {
        headers: {
          'x-internal-secret': process.env.INTERNAL_API_SECRET!
        }
      }
    );
    
    if (!response.ok) {
      return NextResponse.json(
        { error: 'Unable to verify subscription' },
        { status: 500 }
      );
    }
    
    const data = await response.json();
    subscriptionStatus = data.status;
    
    // Cache in Edge Config for 5 minutes
    // Note: Edge Config updates are async, this is just best-effort caching
  }
  
  // Block if expired
  if (['EXPIRED', 'CANCELED', 'PAUSED'].includes(subscriptionStatus)) {
    return NextResponse.json(
      {
        error: 'Store unavailable',
        code: 'SUBSCRIPTION_EXPIRED'
      },
      { status: 402 }
    );
  }
  
  return NextResponse.next();
}
```

**Step 3: Internal Subscription Check API (For Edge)**

```typescript
// File: src/app/api/internal/check-subscription/[storeId]/route.ts

import { NextResponse } from 'next/server';
import prismadb from '@/lib/prismadb';
import { headers } from 'next/headers';

export async function GET(
  req: Request,
  { params }: { params: Promise<{ storeId: string }> }
) {
  // Verify internal request (prevent public access)
  const headersList = await headers();
  const secret = headersList.get('x-internal-secret');
  
  if (secret !== process.env.INTERNAL_API_SECRET) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  
  const { storeId } = await params;
  
  // Query subscription status
  const store = await prismadb.store.findUnique({
    where: { id: storeId },
    select: {
      user: {
        select: {
          subscription: {
            select: { status: true }
          }
        }
      }
    }
  });
  
  if (!store) {
    return NextResponse.json({ status: 'NOT_FOUND' }, { status: 404 });
  }
  
  const status = store.user.subscription?.status || 'NONE';
  
  return NextResponse.json({ status });
}
```

### ✅ Middleware Pattern - Pros & Cons

**Advantages:**
1. ✅ **Centralized**: Single place for all access control logic
2. ✅ **Early Blocking**: Request never reaches API handlers if denied
3. ✅ **Performance**: Can run at edge globally (if using Vercel/Cloudflare)
4. ✅ **Consistent**: All APIs protected with same logic
5. ✅ **Headers**: Can add metadata for downstream handlers

**Disadvantages:**
1. ❌ **Edge Limitations**: Can't query database directly in edge runtime
2. ❌ **Debugging**: Harder to debug (runs before handlers)
3. ❌ **Flexibility**: Less granular control per endpoint
4. ❌ **Cold Starts**: Middleware runs on every request (no handler-level optimization)

**Best For:**
- ✅ High-traffic apps needing edge performance
- ✅ When all APIs have same access rules
- ✅ When you want centralized security

## Rate Limiting Strategies

### 🏢 Industry Implementations

#### **1. Token Bucket Algorithm (Stripe)**

```
┌─────────────────────────────────────────────────────────┐
│              Token Bucket Rate Limiting                  │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  Concept:                                                │
│  - Bucket holds N tokens                                 │
│  - Tokens refill at constant rate (R tokens/second)     │
│  - Each request consumes 1 token                         │
│  - If bucket empty → Request denied (429)                │
│                                                          │
│  Example: 100 req/sec burst, 10 req/sec sustained       │
│                                                          │
│  Time: 0s → Bucket has 100 tokens                       │
│         ↓  User makes 100 requests instantly            │
│         → 100 requests succeed, bucket empty             │
│                                                          │
│  Time: 1s → Bucket refilled with 10 tokens              │
│         ↓  User makes 50 requests                       │
│         → 10 requests succeed, 40 denied (429)           │
│                                                          │
│  Time: 10s → Bucket has 100 tokens (refilled to max)    │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

**Stripe Implementation (Conceptual):**

```ruby
# Stripe's rate limiter (Redis-based)

class TokenBucketRateLimiter
  BUCKET_SIZE = 100       # Max tokens
  REFILL_RATE = 10        # Tokens per second
  
  def initialize(redis, key)
    @redis = redis
    @key = "ratelimit:#{key}"
  end
  
  def allow_request?
    now = Time.now.to_f
    
    # Get current bucket state
    tokens, last_refill = @redis.hmget(@key, 'tokens', 'last_refill')
    tokens = tokens&.to_f || BUCKET_SIZE
    last_refill = last_refill&.to_f || now
    
    # Calculate tokens to add based on elapsed time
    elapsed = now - last_refill
    tokens_to_add = elapsed * REFILL_RATE
    tokens = [tokens + tokens_to_add, BUCKET_SIZE].min
    
    # Try to consume a token
    if tokens >= 1
      tokens -= 1
      
      # Update bucket state
      @redis.hmset(@key, 'tokens', tokens, 'last_refill', now)
      @redis.expire(@key, 3600) # 1 hour TTL
      
      return true
    else
      # Bucket empty - rate limited
      return false
    end
  end
  
  def remaining_tokens
    now = Time.now.to_f
    tokens, last_refill = @redis.hmget(@key, 'tokens', 'last_refill')
    tokens = tokens&.to_f || BUCKET_SIZE
    last_refill = last_refill&.to_f || now
    
    elapsed = now - last_refill
    tokens_to_add = elapsed * REFILL_RATE
    [tokens + tokens_to_add, BUCKET_SIZE].min.to_i
  end
end

# Usage in API handler
def handle_api_request(api_key, request)
  limiter = TokenBucketRateLimiter.new($redis, api_key)
  
  unless limiter.allow_request?
    return {
      status: 429,
      body: {
        error: 'Rate limit exceeded',
        retry_after: 1.0 / REFILL_RATE  # Seconds until next token
      }
    }
  end
  
  # Process request
  response = process_request(request)
  
  # Add rate limit headers
  response.headers['X-RateLimit-Limit'] = BUCKET_SIZE
  response.headers['X-RateLimit-Remaining'] = limiter.remaining_tokens
  
  response
end
```

**Why Token Bucket is Great:**
- ✅ Allows bursts (good UX for legitimate users)
- ✅ Smooth rate limiting over time
- ✅ Simple to implement with Redis

#### **2. Sliding Window Algorithm (GitHub)**

```
┌─────────────────────────────────────────────────────────┐
│           Sliding Window Rate Limiting                   │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  Concept:                                                │
│  - Count requests in last N seconds (sliding window)    │
│  - If count > limit → Deny request                      │
│  - More accurate than fixed window                      │
│                                                          │
│  Example: 5,000 req/hour for GitHub Pro                 │
│                                                          │
│  Timeline:                                               │
│  ┌────┬────┬────┬────┬────┬────┬────┐                  │
│  │ 0  │ 15 │ 30 │ 45 │ 60 │ 75 │ 90 │ (minutes)        │
│  └────┴────┴────┴────┴────┴────┴────┘                  │
│    1K   1K   1K   1K   1K   1K   1K  (requests)         │
│                                                          │
│  At minute 45:                                           │
│  - Last 60 minutes: 45→0 = 4,000 requests               │
│  - Can make 1,000 more                                   │
│                                                          │
│  At minute 60:                                           │
│  - Last 60 minutes: 60→0 = 5,000 requests               │
│  - Limit reached, deny requests                          │
│                                                          │
│  At minute 61:                                           │
│  - Last 60 minutes: 61→1 = 5,000 (minute 0 expired)    │
│  - Still limited                                         │
│                                                          │
│  At minute 75:                                           │
│  - Last 60 minutes: 75→15 = 4,000                       │
│  - Can make 1,000 more                                   │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

**GitHub Implementation (Redis Sorted Set):**

```typescript
// GitHub's sliding window rate limiter

import Redis from 'ioredis';

interface RateLimitResult {
  allowed: boolean;
  remaining: number;
  resetAt: number;
}

class SlidingWindowRateLimiter {
  private redis: Redis;
  private limit: number;
  private windowMs: number;
  
  constructor(redis: Redis, limit: number, windowMs: number) {
    this.redis = redis;
    this.limit = limit;
    this.windowMs = windowMs;
  }
  
  async checkLimit(key: string): Promise<RateLimitResult> {
    const now = Date.now();
    const windowStart = now - this.windowMs;
    const redisKey = `ratelimit:${key}`;
    
    // Use Redis transaction for atomicity
    const multi = this.redis.multi();
    
    // Remove old entries outside window
    multi.zremrangebyscore(redisKey, 0, windowStart);
    
    // Count requests in current window
    multi.zcard(redisKey);
    
    // Add current request
    multi.zadd(redisKey, now, `${now}-${Math.random()}`);
    
    // Set expiry
    multi.expire(redisKey, Math.ceil(this.windowMs / 1000));
    
    const results = await multi.exec();
    const count = results![1][1] as number;
    
    const allowed = count < this.limit;
    const remaining = Math.max(0, this.limit - count - 1);
    const resetAt = now + this.windowMs;
    
    return { allowed, remaining, resetAt };
  }
}

// Usage in GitHub API
export async function GET(req: Request) {
  const userId = req.headers.get('x-user-id');
  
  // Different limits per subscription tier
  const limits = {
    'free': { limit: 60, window: 3600000 },      // 60/hour
    'pro': { limit: 5000, window: 3600000 },     // 5,000/hour
    'enterprise': { limit: 15000, window: 3600000 } // 15,000/hour
  };
  
  const userTier = await getUserTier(userId);
  const { limit, window } = limits[userTier];
  
  const limiter = new SlidingWindowRateLimiter(redis, limit, window);
  const result = await limiter.checkLimit(userId);
  
  if (!result.allowed) {
    return new Response(
      JSON.stringify({
        message: 'API rate limit exceeded',
        documentation_url: 'https://docs.github.com/...'
      }),
      {
        status: 429,
        headers: {
          'X-RateLimit-Limit': limit.toString(),
          'X-RateLimit-Remaining': '0',
          'X-RateLimit-Reset': Math.ceil(result.resetAt / 1000).toString(),
          'Retry-After': Math.ceil((result.resetAt - Date.now()) / 1000).toString()
        }
      }
    );
  }
  
  // Process request
  const response = await handleRequest(req);
  
  // Add rate limit headers
  response.headers.set('X-RateLimit-Limit', limit.toString());
  response.headers.set('X-RateLimit-Remaining', result.remaining.toString());
  response.headers.set('X-RateLimit-Reset', Math.ceil(result.resetAt / 1000).toString());
  
  return response;
}
```

#### **3. Fixed Window Counter (Simple but Effective)**

**Used by:** AWS API Gateway, Many REST APIs

```typescript
// Simple fixed window rate limiter (Redis)

async function checkRateLimit(key: string, limit: number, windowSeconds: number) {
  const now = Math.floor(Date.now() / 1000);
  const window = Math.floor(now / windowSeconds);
  const redisKey = `ratelimit:${key}:${window}`;
  
  // Increment counter
  const count = await redis.incr(redisKey);
  
  // Set expiry on first request
  if (count === 1) {
    await redis.expire(redisKey, windowSeconds * 2); // 2x window for safety
  }
  
  return {
    allowed: count <= limit,
    remaining: Math.max(0, limit - count),
    resetAt: (window + 1) * windowSeconds
  };
}

// Example: 1000 requests per hour
const result = await checkRateLimit(userId, 1000, 3600);
```

**Pros:**
- ✅ Extremely simple to implement
- ✅ Very fast (single Redis INCR operation)
- ✅ Low memory usage

**Cons:**
- ❌ "Reset burst" problem: User can make 1000 requests at 11:59, then 1000 at 12:01
- ❌ Less accurate than sliding window

### 🛠️ Rate Limiting for ShopPulse

```typescript
// File: src/lib/rate-limiter.ts

import { Redis } from '@upstash/redis';

const redis = Redis.fromEnv();

interface RateLimitConfig {
  limit: number;        // Max requests
  windowMs: number;     // Time window in milliseconds
}

// Rate limits per subscription tier
const RATE_LIMITS: Record<string, RateLimitConfig> = {
  'TRIAL': { limit: 100, windowMs: 3600000 },    // 100/hour
  'ACTIVE': { limit: 1000, windowMs: 3600000 },  // 1000/hour
  'EXPIRED': { limit: 0, windowMs: 3600000 }     // No access
};

export async function checkRateLimit(
  storeId: string,
  subscriptionStatus: string
): Promise<{ allowed: boolean; remaining: number; resetAt: number }> {
  
  const config = RATE_LIMITS[subscriptionStatus] || RATE_LIMITS['TRIAL'];
  
  // If limit is 0, deny immediately
  if (config.limit === 0) {
    return { allowed: false, remaining: 0, resetAt: Date.now() + config.windowMs };
  }
  
  const now = Date.now();
  const window = Math.floor(now / config.windowMs);
  const key = `ratelimit:${storeId}:${window}`;
  
  // Increment counter
  const count = await redis.incr(key);
  
  // Set expiry on first request
  if (count === 1) {
    await redis.expire(key, Math.ceil(config.windowMs / 1000));
  }
  
  const allowed = count <= config.limit;
  const remaining = Math.max(0, config.limit - count);
  const resetAt = (window + 1) * config.windowMs;
  
  return { allowed, remaining, resetAt };
}

// Wrapper for GET endpoints with rate limiting
export function withRateLimit(handler: Function) {
  return async (req: Request, context: any) => {
    const params = await context.params;
    const { storeId } = params;
    
    // Get subscription status from header (set by middleware)
    const subscriptionStatus = req.headers.get('x-subscription-status') || 'TRIAL';
    
    // Check rate limit
    const rateLimit = await checkRateLimit(storeId, subscriptionStatus);
    
    if (!rateLimit.allowed) {
      return new Response(
        JSON.stringify({
          error: 'Rate limit exceeded',
          code: 'RATE_LIMIT_EXCEEDED',
          retryAfter: Math.ceil((rateLimit.resetAt - Date.now()) / 1000)
        }),
        {
          status: 429,
          headers: {
            'Content-Type': 'application/json',
            'X-RateLimit-Limit': RATE_LIMITS[subscriptionStatus].limit.toString(),
            'X-RateLimit-Remaining': '0',
            'X-RateLimit-Reset': Math.ceil(rateLimit.resetAt / 1000).toString(),
            'Retry-After': Math.ceil((rateLimit.resetAt - Date.now()) / 1000).toString()
          }
        }
      );
    }
    
    // Process request
    const response = await handler(req, context);
    
    // Add rate limit headers
    response.headers.set('X-RateLimit-Limit', RATE_LIMITS[subscriptionStatus].limit.toString());
    response.headers.set('X-RateLimit-Remaining', rateLimit.remaining.toString());
    response.headers.set('X-RateLimit-Reset', Math.ceil(rateLimit.resetAt / 1000).toString());
    
    return response;
  };
}
```

## Separate CDN/Domain Approach

### 🏢 Industry Example: Vercel Deployments

```
┌─────────────────────────────────────────────────────────┐
│           Vercel Multi-Domain Architecture               │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  Each Deployment Gets Unique URL:                        │
│  ├── mystore-abc123.vercel.app (Active)                 │
│  ├── mystore-xyz789.vercel.app (Preview)                │

│  └── If subscription expires:                            │
│      ├── DNS record updated (points to "paused" page)   │

│      └── CDN serves static "account suspended" page     │
│                                                          │
│  Reactivation:                                           │
│  └── Payment successful:                                 │
│      ├── DNS updated (points back to deployment)        │
│      ├── Instant restoration (seconds)                  │
│      └── No re-deployment needed                        │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

**Implementation Concept:**

```typescript
// Conceptual: Vercel's deployment management

interface Deployment {
  id: string;
  url: string;
  customDomain?: string;
  status: 'active' | 'paused' | 'cancelled';
}

async function handleSubscriptionExpired(userId: string) {
  // Get all user deployments
  const deployments = await db.deployments.findMany({
    where: { userId }
  });
  
  for (const deployment of deployments) {
    // Update DNS to point to "paused" page
    await updateDNS({
      domain: deployment.customDomain,
      target: 'paused.vercel-status.com',
      type: 'CNAME'
    });
    
    // Update deployment status
    await db.deployments.update({
      where: { id: deployment.id },
      data: { status: 'paused' }
    });
    
    // Purge CDN cache
    await purgeCDN(deployment.url);
  }
}

async function handleSubscriptionReactivated(userId: string) {
  const deployments = await db.deployments.findMany({
    where: { userId, status: 'paused' }
  });
  
  for (const deployment of deployments) {
    // Restore DNS
    await updateDNS({
      domain: deployment.customDomain,
      target: `${deployment.id}.vercel-dns.com`,
      type: 'CNAME'
    });
    
    // Update status
    await db.deployments.update({
      where: { id: deployment.id },
      data: { status: 'active' }
    });
    
    // Purge CDN cache
    await purgeCDN(deployment.url);
  }
}
```

## Interview-Ready Summary

### 🎤 "Explain API Access Control Patterns"

**Answer Template:**

"There are 5 main patterns for API access control in subscription-based SaaS:

**1. API Key Based** (Stripe, AWS)
- Each client gets unique key
- Key checked on every request
- Pros: Granular control, detailed analytics
- Cons: Key management overhead, user friction
- Best for: Multi-tenant platforms

**2. Store Status Flag** (Shopify, Wix)
- Database flag controls access
- Cached for performance
- Pros: Simple, fast, transparent
- Cons: Less granular
- Best for: Single-tenant stores

**3. Middleware Check** (Vercel, Next.js)
- Centralized check before handlers
- Can run at edge globally
- Pros: Fast, consistent, early blocking
- Cons: Less flexible per-endpoint
- Best for: High-traffic apps

**4. Rate Limiting** (GitHub, Twitter)
- Controls request volume
- Tiered by subscription
- Pros: Prevents abuse, enables tiers
- Cons: Requires Redis, complexity
- Best for: Pay-per-use models

**5. Separate CDN/Domain** (Vercel deployments)
- Each customer isolated domain
- DNS controls access
- Pros: Ultra-fast, secure
- Cons: Very complex, expensive
- Best for: Enterprise scale

In my ShopPulse project, I'd use **Store Status Flag** because:
- Simple store-to-dashboard architecture
- Single owner per store
- Easy to implement and debug
- 5-minute cache gives <10ms performance
- Transparent to end customers"

### 🎤 "How Would You Handle Rate Limiting?"

**Answer:**

"I'd implement a **token bucket** or **sliding window** algorithm using Redis:

```typescript
// Pseudo-code
async function checkRateLimit(userId, tier) {
  const limits = {
    free: 100/hour,
    pro: 1000/hour,
    enterprise: 10000/hour
  };
  
  const key = `ratelimit:${userId}:${currentHour}`;
  const count = await redis.incr(key);
  
  if (count > limits[tier]) {
    return {
      allowed: false,
      retryAfter: 3600 - (Date.now() % 3600)
    };
  }
  
  return {
    allowed: true,
    remaining: limits[tier] - count
  };
}
```

**Key considerations:**
1. Use Redis for speed (< 1ms lookups)
2. Add proper headers (X-RateLimit-Remaining)
3. Return 429 with Retry-After
4. Different limits per subscription tier
5. Consider burst allowance (token bucket)
6. Monitor for abuse patterns"

### 🎤 "API Key vs JWT - Which is Better?"

**Answer:**

"It depends on use case:

**API Keys** (Stripe, AWS):
- ✅ Long-lived, revocable
- ✅ Different keys per environment
- ✅ Simple for public APIs
- ❌ Must be stored securely
- ❌ Requires database lookup

**JWT** (Auth0, Clerk):
- ✅ Stateless (no DB lookup)
- ✅ Contains claims (user, subscription)
- ✅ Short-lived (more secure)
- ❌ Can't revoke before expiry
- ❌ Larger size (payload)

**Hybrid Approach** (Best of both):
- JWT for user authentication
- API key for server-to-server
- Refresh tokens for renewals

For ShopPulse, I'd use:
- **Clerk JWT** for dashboard (user auth)
- **Store Status Flag** for store APIs (no keys needed)
- **API Keys** if adding 3rd-party integrations later"

### 🔀 Combining Patterns (Production-Ready)

```typescript
// File: src/lib/api-access-control.ts

/**
 * Comprehensive access control combining multiple patterns
 */

import { checkStoreStatus } from './store-status-service';
import { validateApiKey } from './api-key-service';
import { checkRateLimit } from './rate-limiter';

interface AccessControlResult {
  allowed: boolean;
  reason?: string;
  headers?: Record<string, string>;
}

export async function comprehensiveAccessCheck(
  req: Request,
  storeId: string
): Promise<AccessControlResult> {
  
  // 1. Check API Key (if provided)
  const apiKey = req.headers.get('x-api-key');
  if (apiKey) {
    const keyValidation = await validateApiKey(apiKey);
    
    if (!keyValidation.valid) {
      return {
        allowed: false,
        reason: 'Invalid API key'
      };
    }
    
    // Verify key matches storeId
    if (keyValidation.storeId !== storeId) {
      return {
        allowed: false,
        reason: 'API key does not match store'
      };
    }
  }
  
  // 2. Check Store Status (subscription)
  const storeStatus = await checkStoreStatus(storeId);
  
  if (!storeStatus.accessible) {
    return {
      allowed: false,
      reason: storeStatus.reason
    };
  }
  
  // 3. Check Rate Limit
  const subscriptionStatus = storeStatus.status;
  const rateLimit = await checkRateLimit(storeId, subscriptionStatus);
  
  if (!rateLimit.allowed) {
    return {
      allowed: false,
      reason: 'Rate limit exceeded',
      headers: {
        'X-RateLimit-Remaining': '0',
        'Retry-After': Math.ceil((rateLimit.resetAt - Date.now()) / 1000).toString()
      }
    };
  }
  
  // All checks passed
  return {
    allowed: true,
    headers: {
      'X-Subscription-Status': subscriptionStatus,
      'X-RateLimit-Remaining': rateLimit.remaining.toString()
    }
  };
}

// Wrapper for GET endpoints
export function withComprehensiveAccessCheck(handler: Function) {
  return async (req: Request, context: any) => {
    const params = await context.params;
    const { storeId } = params;
    
    const accessCheck = await comprehensiveAccessCheck(req, storeId);
    
    if (!accessCheck.allowed) {
      return new Response(
        JSON.stringify({
          error: 'Access denied',
          reason: accessCheck.reason
        }),
        {
          status: 402,
          headers: {
            'Content-Type': 'application/json',
            ...accessCheck.headers
          }
        }
      );
    }
    
    // Process request
    const response = await handler(req, context);
    
    // Add access control headers
    if (accessCheck.headers) {
      Object.entries(accessCheck.headers).forEach(([key, value]) => {
        response.headers.set(key, value);
      });
    }
    
    return response;
  };
}
```
