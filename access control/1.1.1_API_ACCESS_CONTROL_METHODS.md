**Step 1: API Key Generation Service**

```typescript
// File: src/lib/api-key-service.ts

import crypto from 'crypto';
import prismadb from './prismadb';

export async function generateApiKey(storeId: string, name: string) {
  // Generate secure random key
  const randomBytes = crypto.randomBytes(32);
  const key = `sk_live_${randomBytes.toString('base64url').substring(0, 32)}`;
  
  // Hash for storage (never store plain key)
  const keyHash = crypto.createHash('sha256').update(key).digest('hex');
  
  // Store in database
  const apiKey = await prismadb.apiKey.create({
    data: {
      key: keyHash,
      keyPrefix: key.substring(0, 16), // sk_live_abc12345
      name,
      storeId,
      rateLimit: 1000, // Default: 1000 req/hour
    }
  });
  
  // Return plain key ONLY once (user must save it)
  return {
    id: apiKey.id,
    key, // Plain text - only shown once
    keyPrefix: apiKey.keyPrefix,
    name: apiKey.name,
    createdAt: apiKey.createdAt
  };
}

export async function validateApiKey(key: string): Promise<{
  valid: boolean;
  storeId?: string;
  rateLimitRemaining?: number;
  subscriptionValid?: boolean;
}> {
  // Hash incoming key
  const keyHash = crypto.createHash('sha256').update(key).digest('hex');
  
  // Lookup in database
  const apiKey = await prismadb.apiKey.findUnique({
    where: { key: keyHash },
    include: {
      store: {
        include: {
          user: {
            include: { subscription: true }
          }
        }
      }
    }
  });
  
  // Check if key exists and not revoked
  if (!apiKey || apiKey.revokedAt) {
    return { valid: false };
  }
  
  // Check if expired
  if (apiKey.expiresAt && apiKey.expiresAt < new Date()) {
    return { valid: false };
  }
  
  // Check subscription status
  const subscription = apiKey.store.user.subscription;
  const subscriptionValid = subscription && 
    ['ACTIVE', 'COMPLETING', 'PAST_DUE'].includes(subscription.status);
  
  if (!subscriptionValid) {
    return {
      valid: false,
      storeId: apiKey.storeId,
      subscriptionValid: false
    };
  }
  
  // Check rate limit (using Redis for speed)
  const hour = new Date().getHours();
  const rateLimitKey = `ratelimit:${apiKey.id}:${hour}`;
  const currentUsage = await redis.incr(rateLimitKey);
  await redis.expire(rateLimitKey, 3600); // 1 hour TTL
  
  if (currentUsage > apiKey.rateLimit) {
    return {
      valid: false,
      storeId: apiKey.storeId,
      rateLimitRemaining: 0
    };
  }
  
  // Update last used timestamp (async, don't await)
  prismadb.apiKey.update({
    where: { id: apiKey.id },
    data: { lastUsedAt: new Date() }
  }).catch(console.error);
  
  return {
    valid: true,
    storeId: apiKey.storeId,
    rateLimitRemaining: apiKey.rateLimit - currentUsage,
    subscriptionValid: true
  };
}
```

**Step 2: Middleware Integration**

```typescript
// File: src/middleware.ts

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { validateApiKey } from '@/lib/api-key-service';

export async function middleware(req: NextRequest) {
  // Check if API route
  if (req.nextUrl.pathname.startsWith('/api/')) {
    
    // Extract API key from header
    const apiKey = req.headers.get('x-shoppulse-key') || 
                   req.headers.get('authorization')?.replace('Bearer ', '');
    
    if (!apiKey) {
      return NextResponse.json(
        { error: 'API key required', code: 'NO_API_KEY' },
        { status: 401 }
      );
    }
    
    // Validate API key
    const validation = await validateApiKey(apiKey);
    
    if (!validation.valid) {
      if (validation.subscriptionValid === false) {
        return NextResponse.json(
          {
            error: 'Subscription expired',
            code: 'SUBSCRIPTION_EXPIRED',
            message: 'Store owner subscription has expired'
          },
          { status: 402 }
        );
      }
      
      if (validation.rateLimitRemaining === 0) {
        return NextResponse.json(
          {
            error: 'Rate limit exceeded',
            code: 'RATE_LIMIT_EXCEEDED',
            retry_after: 3600
          },
          { status: 429 }
        );
      }
      
      return NextResponse.json(
        { error: 'Invalid API key', code: 'INVALID_KEY' },
        { status: 401 }
      );
    }
    
    // Verify storeId matches (prevent key reuse across stores)
    const storeIdInUrl = req.nextUrl.pathname.split('/')[2];
    if (storeIdInUrl !== validation.storeId) {
      return NextResponse.json(
        { error: 'API key does not match store', code: 'STORE_MISMATCH' },
        { status: 403 }
      );
    }
    
    // Add rate limit headers
    const response = NextResponse.next();
    response.headers.set('X-RateLimit-Remaining', 
      validation.rateLimitRemaining?.toString() || '0'
    );
    return response;
  }
  
  return NextResponse.next();
}
```

### ✅ API Key Pattern - Pros & Cons

**Advantages:**
1. ✅ **Fine-grained Control**: Different keys for different environments (dev/prod)
2. ✅ **Easy Revocation**: Revoke compromised key without affecting others
3. ✅ **Rate Limiting**: Apply different limits per key
4. ✅ **Analytics**: Track usage per key for billing/monitoring
5. ✅ **Security**: Keys can expire, be rotated, have scopes

**Disadvantages:**
1. ❌ **Complexity**: More infrastructure (key generation, storage, validation)
2. ❌ **User Experience**: Users must manage keys (copy, paste, secure)
3. ❌ **Key Leakage Risk**: If exposed in frontend code or Git
4. ❌ **Performance**: Extra database lookup on every request (mitigated by caching)

**Best For:**
- ✅ SaaS platforms with multiple integrations
- ✅ APIs consumed by multiple clients/apps
- ✅ When you need detailed usage analytics
- ✅ When rate limiting per client is critical

**Not Ideal For:**
- ❌ Simple store-to-dashboard architecture (overhead)
- ❌ Non-technical users (key management friction)
- ❌ If performance is critical (adds latency)

**Shopify Implementation Details:**

```ruby
# Shopify's Rails backend (conceptual)

# Storefront Request
class StorefrontController < ApplicationController
  before_action :check_store_status
  
  def check_store_status
    @shop = Shop.find_by(domain: request.host)
    
    # Check store status flag
    unless @shop.status == 'active' || @shop.status == 'trial'
      render 'errors/store_closed', status: 503
      return false
    end
    
    # Check subscription (cached for 5 minutes)
    subscription_status = Rails.cache.fetch(
      "shop:#{@shop.id}:subscription",
      expires_in: 5.minutes
    ) do
      @shop.subscription&.status
    end
    
    unless ['active', 'trial', 'frozen'].include?(subscription_status)
      @shop.update(status: 'closed', store_access: false)
      render 'errors/store_closed', status: 503
      return false
    end
  end
end

# Admin Dashboard Request
class Admin::DashboardController < ApplicationController
  before_action :check_subscription
  
  def check_subscription
    @shop = current_shop
    subscription = @shop.subscription
    
    case subscription.status
    when 'active', 'trial'
      # Full access
      return true
    when 'frozen'
      # Limited access with warning
      flash.now[:warning] = "Payment failed. Please update payment method."
      return true
    when 'closed', 'cancelled'
      # Blocked access
      redirect_to subscription_path, 
        alert: "Subscription required to access dashboard"
      return false
    end
  end
end

# Subscription Webhook Handler
class SubscriptionWebhookController < ApplicationController
  def charge_failed
    shop = Shop.find(params[:shop_id])
    subscription = shop.subscription
    
    # Update status
    subscription.update(status: 'frozen')
    
    # Store still accessible (grace period)
    shop.update(status: 'frozen')
    
    # Invalidate cache
    Rails.cache.delete("shop:#{shop.id}:subscription")
    
    # Send email notification
    SubscriptionMailer.payment_failed(shop).deliver_later
  end
  
  def subscription_cancelled
    shop = Shop.find(params[:shop_id])
    
    # Close store immediately
    shop.update(status: 'closed', store_access: false)
    
    # Invalidate cache
    Rails.cache.delete("shop:#{shop.id}:subscription")
    
    # Notify owner
    SubscriptionMailer.store_closed(shop).deliver_later
  end
end
```

**Key Insights from Shopify:**
- **Single Source of Truth**: `shop.status` flag controls everything
- **Grace Periods**: 'frozen' status gives 7 days before 'closed'
- **Caching**: Subscription status cached for 5 minutes to reduce DB load
- **Instant Invalidation**: Webhooks clear cache immediately
- **Customer Experience**: Shows "Store Unavailable" (not broken page)

```javascript
// Wix's Node.js backend (conceptual)

// CDN Edge Worker (Cloudflare Workers)
addEventListener('fetch', event => {
  event.respondWith(handleRequest(event.request))
})

async function handleRequest(request) {
  const url = new URL(request.url)
  const siteSlug = url.hostname.split('.')[0] // 'mystore' from mystore.wixsite.com
  
  // Check CDN cache first
  const cacheKey = `site:${siteSlug}:status`
  let siteStatus = await EDGE_KV.get(cacheKey)
  
  if (!siteStatus) {
    // Cache miss - fetch from origin
    const response = await fetch(`https://api.wix.com/_internal/site-status/${siteSlug}`)
    const data = await response.json()
    siteStatus = data.status
    
    // Cache at edge for 10 minutes
    await EDGE_KV.put(cacheKey, siteStatus, { expirationTtl: 600 })
  }
  
  // Route based on status
  if (siteStatus === 'paused' || siteStatus === 'unpublished') {
    return new Response(PAUSED_HTML, {
      status: 503,
      headers: { 'Content-Type': 'text/html' }
    })
  }
  
  // Site active - fetch from origin
  return fetch(request)
}

// Origin Server (Node.js)
app.get('/site/:slug', async (req, res) => {
  const site = await db.query(`
    SELECT 
      s.id, s.status, s.premium_expires_at,
      sp.status as subscription_status
    FROM sites s
    LEFT JOIN subscription_plans sp ON s.premium_plan_id = sp.id
    WHERE s.slug = $1
  `, [req.params.slug])
  
  // Compute real-time status
  const now = new Date()
  const isActive = 
    site.status === 'published' &&
    site.premium_expires_at > now &&
    ['active', 'trialing'].includes(site.subscription_status)
  
  if (!isActive) {
    // Update database (async)
    db.query(`
      UPDATE sites 
      SET status = 'paused', is_active = false 
      WHERE id = $1
    `, [site.id]).catch(console.error)
    
    return res.status(503).render('site-paused')
  }
  
  // Serve site content
  res.render('site', { site })
})

// Subscription Webhook
app.post('/webhooks/subscription/:event', async (req, res) => {
  const { site_id, subscription_status } = req.body
  
  if (subscription_status === 'expired') {
    // Update site status immediately
    await db.query(`
      UPDATE sites 
      SET status = 'paused', is_active = false 
      WHERE id = $1
    `, [site_id])
    
    // Purge CDN cache globally
    await purgeCloudflareCache(`site:${site.slug}:status`)

    await sendEmail({
```

**Step 2: Store Access Service (Simplified)**

```typescript
// File: src/lib/store-status-service.ts

import prismadb from './prismadb';

interface StoreStatusResult {
  accessible: boolean;
  status: 'ACTIVE' | 'SUSPENDED' | 'CLOSED' | 'NOT_FOUND';
  reason?: string;
}

// In-memory cache
const statusCache = new Map<string, { result: StoreStatusResult; timestamp: number }>();
const CACHE_TTL = 5 * 60 * 1000; // 5 minutes

export async function checkStoreStatus(storeId: string): Promise<StoreStatusResult> {
  // Check cache
  const cached = statusCache.get(storeId);
  if (cached && Date.now() - cached.timestamp < CACHE_TTL) {
    return cached.result;
  }
  
  // Query database (single query with joins)
  const store = await prismadb.store.findUnique({
    where: { id: storeId },
    include: {
      user: {
        include: { subscription: true }
      }
    }
  });
  
  if (!store) {
    const result: StoreStatusResult = {
      accessible: false,
      status: 'NOT_FOUND',
      reason: 'Store does not exist'
    };
    cacheResult(storeId, result);
    return result;
  }
  
  // Compute accessibility
  const user = store.user;
  const subscription = user.subscription;
  
  // Check 1: Store status flag
  if (store.status === 'CLOSED') {
    const result: StoreStatusResult = {
      accessible: false,
      status: 'CLOSED',
      reason: 'Store has been closed due to subscription expiry'
    };
    cacheResult(storeId, result);
    return result;
  }
  
  // Check 2: User active flag
  if (!user.isActive) {
    const result: StoreStatusResult = {
      accessible: false,
      status: 'CLOSED',
      reason: 'User account is inactive'
    };
    cacheResult(storeId, result);
    return result;
  }
  
  // Check 3: Subscription status
  if (!subscription || !['ACTIVE', 'COMPLETING', 'PAST_DUE'].includes(subscription.status)) {
    // Update store status to CLOSED (async)
    prismadb.store.update({
      where: { id: storeId },
      data: { status: 'CLOSED' }
    }).catch(console.error);
    
    const result: StoreStatusResult = {
      accessible: false,
      status: 'CLOSED',
      reason: 'Subscription expired or cancelled'
    };
    cacheResult(storeId, result);
    return result;
  }
  
  // All checks passed
  const result: StoreStatusResult = {
    accessible: true,
    status: store.status === 'SUSPENDED' ? 'SUSPENDED' : 'ACTIVE'
  };
  cacheResult(storeId, result);
  return result;
}

// Cache helpers
function cacheResult(storeId: string, result: StoreStatusResult) {
  statusCache.set(storeId, { result, timestamp: Date.now() });
}

export function invalidateStoreStatusCache(storeId: string) {
  statusCache.delete(storeId);
}

export async function invalidateUserStoresCache(userId: string) {
  const stores = await prismadb.store.findMany({
    where: { userId },
    select: { id: true }
  });
  stores.forEach(store => statusCache.delete(store.id));
}
```

**Step 3: API Wrapper (Using Status Check)**

```typescript
// File: src/lib/api-with-status-check.ts

import { NextResponse } from 'next/server';
import { checkStoreStatus } from './store-status-service';

type ApiHandler = (req: Request, context: any) => Promise<Response>;

export function withStoreStatusCheck(handler: ApiHandler): ApiHandler {
  return async (req: Request, context: any) => {
    const params = await context.params;
    const { storeId } = params;
    
    // Check store status
    const statusCheck = await checkStoreStatus(storeId);
    
    if (!statusCheck.accessible) {
      return NextResponse.json(
        {
          error: 'Store unavailable',
          status: statusCheck.status,
          reason: statusCheck.reason,
          code: 'STORE_CLOSED'
        },
        { status: 402 }
      );
    }
    
    // If suspended, add warning header (but allow access)
    const response = await handler(req, context);
    
    if (statusCheck.status === 'SUSPENDED') {
      response.headers.set('X-Store-Status', 'suspended');
      response.headers.set('X-Warning', 'Store in grace period');
    }
    
    return response;
  };
}
```

```typescript
      where: { userId: subscription.userId },

    });
  });

}

async function handleSubscriptionCharged(webhookData: any) {
  const subscription = await prismadb.subscription.findUnique({
    where: { razorpaySubscriptionId: webhookData.subscription.id }
  });
  
  await prismadb.$transaction(async (tx) => {
    // Update subscription
    await tx.subscription.update({
      where: { id: subscription.id },
      data: { status: 'ACTIVE' }
    });
    
    // Reactivate user
    await tx.user.update({
      where: { id: subscription.userId },
      data: { isActive: true }
    });
    
    // ⭐ Reopen all stores
    await tx.store.updateMany({
      where: { userId: subscription.userId },
      data: { status: 'ACTIVE' }
    });
  });

  await invalidateUserStoresCache(subscription.userId);
```

### ✅ Store Status Flag - Pros & Cons

**Advantages:**
1. ✅ **Simplicity**: Single database flag controls everything
2. ✅ **Performance**: One query with joins (no multiple lookups)
3. ✅ **Consistency**: All parts of app check same flag
4. ✅ **Auditability**: Easy to see store status history
5. ✅ **Scalability**: Minimal overhead per request

**Disadvantages:**
1. ❌ **Less Granular**: Can't have different access levels per API
2. ❌ **Database Dependency**: Must query DB (mitigated by caching)
3. ❌ **Status Sync**: Must update flag in multiple places (webhooks, crons)

**Best For:**
- ✅ **Your ShopPulse App** - Perfect fit! Simple store-to-dashboard architecture
- ✅ Single-tenant stores (one owner, one subscription)
- ✅ When simplicity and performance matter more than granularity

**Vercel Example Implementation:**

```typescript
// middleware.ts (runs at Vercel edge)

import { NextRequest, NextResponse } from 'next/server';
import { verifyToken } from '@/lib/auth-edge'; // Edge-compatible

export async function middleware(req: NextRequest) {
  const token = req.cookies.get('session')?.value;
  
  // Verify JWT at edge (no database lookup)
  const user = await verifyToken(token);
  
  if (!user) {
    return NextResponse.redirect(new URL('/login', req.url));
  }
  
  // Check subscription (stored in JWT claims for speed)
  if (user.subscription !== 'active' && req.nextUrl.pathname.startsWith('/dashboard')) {
    return NextResponse.redirect(new URL('/upgrade', req.url));
  }
  
  // Add user info to headers for downstream consumption
  const response = NextResponse.next();
  response.headers.set('x-user-id', user.id);
  response.headers.set('x-subscription', user.subscription);
  
  return response;
}

export const config = {
  matcher: [
    '/dashboard/:path*',
    '/api/:path*'
  ]
};
```

**Why Vercel Uses This:**
- **Speed**: Middleware runs at edge (globally distributed)
- **No DB Lookups**: JWT contains subscription info (verified cryptographically)
- **Scalability**: Handles millions of requests without backend load
- **Security**: Request never reaches backend if unauthorized
