## 🔀 Hybrid Approach (Clerk + Redis)

### **Architecture**

```
┌─────────────────────────────────────────────────────────┐
│         Hybrid: Clerk + Redis Architecture               │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  ✅ Keep Clerk for:                                      │
│     - User authentication (OAuth, magic links)          │
│     - User management UI                                │
│     - Session management                                │
│     - Profile/metadata                                  │
│                                                          │
│  ✅ Add Redis for:                                       │
│     - Subscription status caching                       │
│     - Rate limiting state                               │
│     - Edge-level authorization                          │
│     - Instant invalidation                              │
│                                                          │
│  Flow:                                                   │
│  1. User logs in via Clerk                              │
│  2. On login, cache subscription in Redis               │
│  3. Middleware: Verify with Clerk (auth) ✅             │
│  4. Middleware: Check Redis (subscription) ✅           │
│  5. Handler: Business logic only                        │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

### **Implementation**

**Setup Redis:**

```typescript
// File: src/lib/redis.ts

import { Redis } from '@upstash/redis';

// ⭐ Upstash Redis works in edge runtime
export const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!
});

// Helper: Cache user subscription
export async function cacheUserSubscription(userId: string) {
  const user = await prismadb.user.findUnique({
    where: { id: userId },
    select: {
      isActive: true,
      gracePeriod: true,
      subscription: {
        select: {
          status: true,
          currentPeriodEnd: true,
          planId: true
        }
      }
    }
  });
  
  if (!user) return null;
  
  const cacheData = {
    isActive: user.isActive,
    gracePeriod: user.gracePeriod,
    subscriptionStatus: user.subscription?.status || 'NONE',
    subscriptionExpiry: user.subscription?.currentPeriodEnd?.toISOString(),
    planId: user.subscription?.planId,
    cachedAt: new Date().toISOString()
  };
  
  // Cache for 5 minutes
  await redis.setex(
    `user:${userId}:subscription`,
    300, // 5 minutes
    JSON.stringify(cacheData)
  );
  
  return cacheData;
}

// Helper: Get cached subscription
export async function getCachedSubscription(userId: string) {
  const cached = await redis.get(`user:${userId}:subscription`);
  
  if (!cached) return null;
  
  return typeof cached === 'string' ? JSON.parse(cached) : cached;
}

// Helper: Invalidate subscription cache
export async function invalidateSubscriptionCache(userId: string) {
  await redis.del(`user:${userId}:subscription`);
}
```

**Updated Middleware:**

```typescript
// File: src/middleware.ts

import { clerkMiddleware } from '@clerk/nextjs/server';
import { NextResponse } from 'next/server';
import { getCachedSubscription, cacheUserSubscription } from '@/lib/redis';

export default clerkMiddleware(async (auth, req) => {
  // 1. Clerk authentication (keeps OAuth, user management, etc.)
  const { userId } = await auth();
  
  if (!userId) {
    return NextResponse.redirect(new URL('/login', req.url));
  }
  
  // 2. ⭐ Check subscription from Redis (edge-compatible!)
  let subscription = await getCachedSubscription(userId);
  
  // 3. If not cached, fetch and cache
  if (!subscription) {
    subscription = await cacheUserSubscription(userId);
  }
  
  // 4. Authorization check (at edge!)
  if (!subscription || 
      !subscription.isActive || 
      subscription.subscriptionStatus === 'EXPIRED') {
    return NextResponse.json(
      {
        error: 'Subscription expired',
        code: 'SUBSCRIPTION_EXPIRED',
        status: subscription?.subscriptionStatus
      },
      { status: 402 }
    );
  }
  
  // 5. Add to headers for handler
  const response = NextResponse.next();
  response.headers.set('X-User-Id', userId);
  response.headers.set('X-Subscription-Status', subscription.subscriptionStatus);
  
  if (subscription.gracePeriod) {
    response.headers.set('X-Grace-Period', 'true');
  }
  
  return response;
});
```

```typescript
// File: src/app/api/webhook/razorpay/subscription/route.ts

import { invalidateSubscriptionCache } from '@/lib/redis';

async function handleSubscriptionExpired(subscription: any) {
  // 1. Update PostgreSQL
  await prismadb.subscription.update({
    where: { id: subscription.id },
    data: { status: 'EXPIRED' }
  });
  
  await prismadb.user.update({
    where: { id: subscription.userId },
    data: { isActive: false }
  });
  
  // 2. ⭐ Invalidate Redis cache (instant effect!)
  await invalidateSubscriptionCache(subscription.userId);
  
  // Next request will:
  // - Check Redis (not found)
  // - Fetch from PostgreSQL
  // - Get updated status (EXPIRED)
  // - Block immediately
  
  console.log(`✅ Invalidated cache for user ${subscription.userId}`);
}

async function handleSubscriptionActivated(subscription: any) {
  // Update database
  await prismadb.subscription.update({
    where: { id: subscription.id },
    data: { status: 'ACTIVE' }
  });
  
  await prismadb.user.update({
    where: { id: subscription.userId },
    data: { isActive: true }
  });
  
  // ⭐ Invalidate cache (next request gets fresh ACTIVE status)
  await invalidateSubscriptionCache(subscription.userId);
}
```

```typescript
      });
      
      // Additional: Check origin
      const origin = req.headers.get('origin');
      if (origin && !allowedOrigins.includes(origin)) {
        return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
      }
    `
  },

      "✅ Verify signature on every request",

      "✅ Rotate secrets periodically"

      if (!user || !validPassword) {
```

```typescript
const authorizationSecurity = {
  // 1. Privilege escalation
  privilegeEscalation: {
    threat: "User accesses resources beyond their subscription",
    
    protection: [
      "✅ Verify subscription on EVERY request",
      "✅ Check in middleware (early blocking)",
      "✅ Double-check in handler (defense in depth)",
      "✅ Log access attempts"
    ],
    
    code: `
      // Middleware: First line of defense
      export default clerkMiddleware(async (auth, req) => {
        const { userId } = await auth();
        const subscription = await getCachedSubscription(userId);
        
        if (subscription.status !== 'ACTIVE') {
          // ⭐ Block immediately, log attempt
          console.warn(\`[AUTH] Blocked expired user: \${userId}\`);
          return NextResponse.json({ error: 'Expired' }, { status: 402 });
        }
        
        return NextResponse.next();
      });
      
      // Handler: Second line of defense
      export async function GET(req: Request) {
        const userId = req.headers.get('x-user-id')!;
        
        // ⭐ Re-verify (in case cache stale or middleware bypassed)
        const user = await prismadb.user.findUnique({
          where: { id: userId },
          select: { isActive: true, subscription: { select: { status: true } } }
        });
        
        if (!user.isActive || user.subscription?.status !== 'ACTIVE') {
          console.error(\`[AUTH] Secondary check failed: \${userId}\`);
          return NextResponse.json({ error: 'Expired' }, { status: 402 });
        }
        
        // Proceed with business logic
      }
    `
  },
  
  // 2. Store access control
  storeAccessControl: {
    threat: "User accesses another user's store",
    
    protection: [
      "✅ Verify store ownership on every request",
      "✅ Use WHERE clause with userId",
      "✅ Never trust storeId from URL alone",
      "✅ Log unauthorized access attempts"
    ],
    
    code: `
      // ❌ BAD: Trust storeId from URL
      const { storeId } = await params;
      const products = await prismadb.product.findMany({
        where: { storeId }  // ⚠️ Any user can access any store!
      });
      
      // ✅ GOOD: Verify ownership
      const userId = req.headers.get('x-user-id')!;
      const { storeId } = await params;
      
      const store = await prismadb.store.findFirst({
        where: { 
          id: storeId,
          userId: userId  // ⭐ Verify ownership
        }
      });
      
      if (!store) {
        console.warn(\`[AUTH] Unauthorized store access: \${userId} → \${storeId}\`);
        return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
      }
      
      // Now safe to query products
      const products = await prismadb.product.findMany({
        where: { storeId: store.id }
      });
    `
  },
  
  // 3. Rate limit bypass
  rateLimitBypass: {
    threat: "User exceeds rate limit by manipulating headers",
    
    protection: [
      "✅ Rate limit by userId (not IP or headers)",
      "✅ Use Redis counters (atomic operations)",
      "✅ Check limit in middleware (before handler)",
      "✅ Return 429 with Retry-After header"

    code: `
      // Rate limiting middleware
      export async function checkRateLimit(userId: string, limit: number) {
        const key = \`ratelimit:\${userId}:\${Date.now() / 60000 | 0}\`;
        
        // ⭐ Atomic increment (no race conditions)
        const current = await redis.incr(key);
        
        if (current === 1) {
          // First request this minute, set expiry
          await redis.expire(key, 60);
        }
        
        if (current > limit) {
          return {
            allowed: false,
            remaining: 0,
            resetAt: new Date(Math.ceil(Date.now() / 60000) * 60000)
          };
        }
        
        return {
          allowed: true,
          remaining: limit - current,
          resetAt: new Date(Math.ceil(Date.now() / 60000) * 60000)
        };
      }
      
      // In middleware
      const { userId } = await auth();
      const rateLimit = await checkRateLimit(userId, 1000);
      
      if (!rateLimit.allowed) {
        return NextResponse.json(
          { error: 'Rate limit exceeded' },
          { 
            status: 429,
            headers: {
              'Retry-After': '60',
              'X-RateLimit-Remaining': '0',
              'X-RateLimit-Reset': rateLimit.resetAt.toISOString()
            }
          }
        );
      }
    `
  }
};
```
