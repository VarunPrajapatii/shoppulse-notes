## 🏗️ Authentication Architecture Overview

### **What is Authentication vs Authorization?**

```
┌─────────────────────────────────────────────────────────┐
│                  Auth System Components                  │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  Authentication (Who are you?)                           │
│  ├── User login with email/password                     │
│  ├── Verify credentials                                 │
│  ├── Issue session/token                                │
│  └── Result: "This is user ID: abc123"                  │
│                                                          │
│  Authorization (What can you do?)                        │
│  ├── Check user's subscription status                   │
│  ├── Check permissions/roles                            │
│  ├── Apply rate limits                                  │
│  └── Result: "User can access resource" OR "Blocked"    │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

### **Core Question for SaaS Apps:**

**How do we enforce subscription-based access control efficiently?**

```
Option 1: Third-party (Clerk, Auth0)
  ↓
  Pros: Easy setup, maintained for you
  Cons: Limited control, can't customize session data

Option 2: In-house (JWT, Sessions)
  ↓
  Pros: Full control, custom session data (subscription status)
  Cons: More work, security responsibility
```

## 🔵 Clerk Authentication (Current Implementation)

### **Architecture Overview**

```
┌─────────────────────────────────────────────────────────┐
│              Clerk Authentication Flow                   │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  1. User Login (Clerk UI)                               │
│     ├── Email/Password                                  │
│     ├── OAuth (Google, GitHub)                          │
│     └── Magic Link                                      │
│                                                          │
│  2. Clerk Creates Session                               │
│     ├── Stores: userId, email, metadata                │
│     ├── Does NOT store: subscription, isActive, etc.    │
│     └── Issues encrypted session token                  │
│                                                          │
│  3. Session Stored in Cookie                            │
│     ├── __session cookie                                │
│     ├── Managed by Clerk                                │
│     └── Automatically refreshed                         │
│                                                          │
│  4. Middleware Verifies Session                         │
│     ├── clerkMiddleware() runs on every request        │
│     ├── Validates session cryptographically             │
│     ├── Provides: userId, sessionId                     │
│     └── Does NOT provide: subscription status           │
│                                                          │
│  5. Handler Queries Database                            │
│     ├── Use userId to fetch user data                   │
│     ├── Include subscription status                     │
│     └── Make authorization decision                     │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

### **Code Implementation**

**Middleware (Edge Runtime):**

```typescript
// File: src/middleware.ts

import { clerkMiddleware } from '@clerk/nextjs/server';
import { NextResponse } from 'next/server';

export default clerkMiddleware(async (auth, req) => {
  // Clerk provides: auth object
  const { userId } = await auth();
  
  // ✅ Can check: Is user logged in?
  if (!userId) {
    return NextResponse.redirect('/login');
  }
  
  // ❌ Cannot check: Is subscription active?
  // Because: Clerk session doesn't include custom data
  
  // ❌ Cannot do: Query database (Prisma doesn't work in edge)
  // const user = await prisma.user.findUnique(...) // ERROR!
  
  // Can only pass through to handler
  return NextResponse.next();
});

export const config = {
  matcher: [
    '/((?!_next|[^?]*\\.(?:html?|css|js(?!on)|jpe?g|webp|png|gif|svg|ttf|woff2?|ico|csv|docx?|xlsx?|zip|webmanifest)).*)',
    '/(api|trpc)(.*)',
  ],
```

**API Handler (Node.js Runtime):**

```typescript
// File: src/app/api/[storeId]/products/route.ts

import { auth } from '@clerk/nextjs/server';
import prismadb from '@/lib/prismadb';
import { NextResponse } from 'next/server';

export async function GET(req: Request, { params }: any) {
  try {
    // 1. Get userId from Clerk
    const { userId } = await auth();
    
    if (!userId) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    
    const { storeId } = await params;
    
    // 2. Query database for subscription status (REQUIRED)
    const store = await prismadb.store.findUnique({
      where: { id: storeId },
      select: {
        id: true,
        user: {
          select: {
            id: true,
            isActive: true,
            subscription: {
              select: { status: true }
            }
          }
        }
      }
    });
    
    // 3. Check subscription (in handler, not middleware)
    if (!store) {
      return NextResponse.json({ error: 'Store not found' }, { status: 404 });
    }
    
    const subscription = store.user.subscription;
    const allowedStatuses = ['ACTIVE', 'COMPLETING', 'PAST_DUE'];
    
    if (!subscription || !allowedStatuses.includes(subscription.status)) {
      return NextResponse.json(
        { 
          error: 'Store unavailable',
          code: 'SUBSCRIPTION_EXPIRED'
        },
        { status: 402 }
      );
    }
    
    // 4. Process request
    const products = await prismadb.product.findMany({
      where: { storeId }
    });
    
    return NextResponse.json(products);
    
  } catch (error) {
    console.error('[PRODUCTS_GET]', error);
    return NextResponse.json({ error: 'Internal error' }, { status: 500 });
  }
}
```

```typescript

interface ClerkSession {
  userId: string;           // ✅ Available
  sessionId: string;        // ✅ Available

// You do NOT get:
```

### **Limitations of Clerk for SaaS**

```typescript
// Problem 1: Can't check subscription in middleware
export default clerkMiddleware(async (auth, req) => {
  const { userId } = await auth();
  
  // ❌ Clerk doesn't provide subscription status
  // ❌ Can't query database (Prisma not edge-compatible)
  // ❌ Must pass request to handler
  
  // Result: Every request hits your database
});

// Problem 2: Database query on every request (without caching)
export async function GET(req: Request) {
  // MUST query database to check subscription
  const user = await prisma.user.findUnique({
    include: { subscription: true }
  });
  
  // This happens on EVERY request if not cached
}

// Problem 3: Can't block at edge
// Request travels: Edge → Your Server → Database → Decision
// With in-house auth: Edge → Decision (blocked before server)
```

## 🟢 In-House JWT Authentication

### **Architecture Overview**

```
┌─────────────────────────────────────────────────────────┐
│            JWT Authentication Flow                       │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  1. User Login (Your API)                               │
│     POST /api/auth/login                                │
│     { email, password }                                 │
│                                                          │
│  2. Verify Credentials                                  │
│     ├── Query database for user                         │
│     ├── Compare password hash (bcrypt)                  │
│     └── If valid: Generate JWT                          │
│                                                          │
│  3. Generate JWT with Claims                            │
│     ├── Standard claims: userId, email, exp             │
│     ├── Custom claims: subscriptionStatus, isActive     │ ⭐
│     └── Sign with secret key                            │
│                                                          │
│  4. Store JWT in Cookie                                 │
│     ├── httpOnly: true (XSS protection)                 │
│     ├── secure: true (HTTPS only)                       │
│     ├── sameSite: 'lax' (CSRF protection)              │
│     └── Token expires in 7 days                         │
│                                                          │
│  5. Middleware Verifies JWT                             │
│     ├── Extract token from cookie                       │
│     ├── Verify signature (cryptographic)                │
│     ├── Decode claims (NO database query!)              │ ⭐
│     └── Check subscription from claims                  │ ⭐
│                                                          │
│  6. Authorization Decision at Edge                      │
│     ├── If subscription EXPIRED → Block (402)           │
│     ├── If token invalid → Redirect to login            │
│     └── If valid → Pass to handler                      │
│                                                          │
└─────────────────────────────────────────────────────────┘
```

### **Code Implementation**

**Login Endpoint (Generate JWT):**

```typescript
// File: src/app/api/auth/login/route.ts

import { NextResponse } from 'next/server';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import prismadb from '@/lib/prismadb';

export async function POST(req: Request) {
  try {
    const { email, password } = await req.json();
    
    // 1. Find user in database
    const user = await prismadb.user.findUnique({
      where: { email },
      include: { subscription: true }
    });
    
    if (!user) {
      return NextResponse.json(
        { error: 'Invalid credentials' },
        { status: 401 }
      );
    }
    
    // 2. Verify password
    const validPassword = await bcrypt.compare(password, user.passwordHash);
    
    if (!validPassword) {
      return NextResponse.json(
        { error: 'Invalid credentials' },
        { status: 401 }
      );
    }
    
    // 3. Generate JWT with custom claims
    const token = jwt.sign(
      {
        // Standard claims
        userId: user.id,
        email: user.email,
        
        // ⭐ Custom claims (subscription data)
        isActive: user.isActive,
        subscriptionStatus: user.subscription?.status || 'NONE',
        subscriptionExpiry: user.subscription?.currentPeriodEnd,
        gracePeriod: user.gracePeriod,
        
        // Token metadata
        iat: Math.floor(Date.now() / 1000), // Issued at
        exp: Math.floor(Date.now() / 1000) + (7 * 24 * 60 * 60) // Expires in 7 days
      },
      process.env.JWT_SECRET!,
      {
        algorithm: 'HS256'
      }
    );
    
    // 4. Set cookie with JWT
    const response = NextResponse.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        name: user.name
      }
    });
    
    response.cookies.set('session', token, {
      httpOnly: true,      // Can't access via JavaScript (XSS protection)
      secure: process.env.NODE_ENV === 'production', // HTTPS only in production
      sameSite: 'lax',     // CSRF protection
      maxAge: 7 * 24 * 60 * 60, // 7 days
      path: '/'
    });
    
    return response;
    
  } catch (error) {
    console.error('[LOGIN_ERROR]', error);
    return NextResponse.json(
      { error: 'Internal error' },
      { status: 500 }
    );
  }
}
```

**Middleware (Edge Runtime with JWT):**

```typescript
// File: src/middleware.ts

import { NextRequest, NextResponse } from 'next/server';
import { jwtVerify } from 'jose'; // ⭐ Edge-compatible JWT library

export async function middleware(req: NextRequest) {
  // 1. Extract JWT from cookie
  const token = req.cookies.get('session')?.value;
  
  if (!token) {
    return NextResponse.redirect(new URL('/login', req.url));
  }
  
  try {
    // 2. Verify JWT (works in edge runtime!)
    const secret = new TextEncoder().encode(process.env.JWT_SECRET);
    const { payload } = await jwtVerify(token, secret);
    
    // 3. ⭐ Check subscription from JWT claims (NO DATABASE QUERY!)
    if (!payload.isActive || payload.subscriptionStatus === 'EXPIRED') {
      return NextResponse.json(
        {
          error: 'Subscription expired',
          code: 'SUBSCRIPTION_EXPIRED',
          message: 'Please renew your subscription to continue'
        },
        { 
          status: 402,
          headers: {
            'X-Subscription-Status': payload.subscriptionStatus as string
          }
        }
      );
    }
    
    // 4. Add user info to headers for downstream handlers
    const response = NextResponse.next();
    response.headers.set('X-User-Id', payload.userId as string);
    response.headers.set('X-Subscription-Status', payload.subscriptionStatus as string);
    
    // 5. Warning for grace period
    if (payload.gracePeriod) {
      response.headers.set('X-Grace-Period', 'true');
    }
    
    return response;
    
  } catch (error) {
    // Token invalid or expired
    console.error('[MIDDLEWARE_JWT_ERROR]', error);
    return NextResponse.redirect(new URL('/login', req.url));
  }
}

export const config = {
  matcher: [
    '/((?!_next|[^?]*\\.(?:html?|css|js(?!on)|jpe?g|webp|png|gif|svg|ttf|woff2?|ico|csv|docx?|xlsx?|zip|webmanifest)).*)',
    '/(api|trpc)(.*)',
  ],
};
```

```typescript
import prismadb from '@/lib/prismadb';

export async function GET(req: Request, { params }: any) {
  try {
    // ✅ User already verified in middleware
    // ✅ Subscription already checked in middleware
    // No need to query user/subscription again!
    
    const userId = req.headers.get('x-user-id')!;

        id: storeId,

      );

    // Fetch products (no subscription check needed - done in middleware!)
    const products = await prismadb.product.findMany({
      where: { storeId }
    });
    
    return NextResponse.json(products);
    
  } catch (error) {
    console.error('[PRODUCTS_GET]', error);
    return NextResponse.json({ error: 'Internal error' }, { status: 500 });
  }
}
```

### **JWT Token Management**

**Refresh Token Pattern:**

```typescript
// Problem: JWT can't be invalidated before expiry
// Solution: Short-lived access token + long-lived refresh token

// Access Token (short-lived: 15 minutes)
const accessToken = jwt.sign(
  {
    userId: user.id,
    subscriptionStatus: user.subscription?.status,
    // ... other claims
  },
  process.env.JWT_SECRET,
  { expiresIn: '15m' } // ⭐ Short expiry
);

// Refresh Token (long-lived: 7 days)
const refreshToken = jwt.sign(
  {
    userId: user.id,
    type: 'refresh'
  },
  process.env.JWT_REFRESH_SECRET,
  { expiresIn: '7d' }
);

// Store refresh token in database (can be revoked)
await prisma.refreshToken.create({
  data: {
    token: refreshToken,
    userId: user.id,
    expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
  }
});

// Client flow:
// 1. Use access token for API requests (15 min)
// 2. When expired, use refresh token to get new access token
// 3. Refresh token queries DB, fetches latest subscription status
// 4. Issue new access token with updated data
```

**Handling Subscription Changes:**

```typescript
// Webhook: Subscription expired
async function handleSubscriptionExpired(userId: string) {
  // 1. Update database
  await prisma.user.update({
    where: { id: userId },
    data: { 
      isActive: false,
      subscription: {
        update: { status: 'EXPIRED' }
      }
    }
  });
  
  // 2. ⚠️ Problem: Existing JWTs still valid for up to 15 minutes!
  // User can still access until token expires naturally
  
  // Solutions:
  
  // Option A: Blacklist (requires database check - defeats JWT purpose)
  await prisma.tokenBlacklist.create({
    data: {
      userId,
      reason: 'Subscription expired',
      expiresAt: new Date(Date.now() + 15 * 60 * 1000) // 15 min
    }
  });
  
  // Option B: Force re-login (redirect to login page)
  // Next request will generate new token with updated status
  
  // Option C: Accept delay (simpler, 15 min max)
  // Trade-off: Immediate invalidation vs simplicity
}

// In middleware (if using blacklist):
export async function middleware(req: NextRequest) {
  const { payload } = await jwtVerify(token, secret);
  
  // Check blacklist (requires database query - adds latency)
  const blacklisted = await redis.get(`blacklist:${payload.userId}`);
  if (blacklisted) {
    return NextResponse.redirect('/login');
  }
  
  // ... rest of middleware
}
```

## 🟡 In-House Session-Based Authentication

### **Architecture Overview**

```
┌─────────────────────────────────────────────────────────┐
│         Session-Based Authentication Flow                │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  1. User Login (Your API)                               │

│  2. Verify Credentials                                  │

│     ├── Store in Redis:                                 │
│     │   Key: session:abc-123-def                        │
│     │   Value: {                                        │
│     │     userId,                                       │
│     │     email,                                        │
│     │     subscriptionStatus,                           │
│     │     isActive                                      │
│     │   }                                               │
│     └── TTL: 7 days                                     │
│                                                          │
│  4. Store Session ID in Cookie                          │
│     └── Cookie: sessionId=abc-123-def                   │
│                                                          │
│  5. Middleware Queries Redis                            │ ⭐
│     ├── Extract sessionId from cookie                   │
│     ├── Query Redis (edge-compatible!)                  │
│     ├── Get subscription status from session            │
│     └── Make authorization decision                     │
```

**Login Endpoint (Create Session):**

```typescript
// File: src/app/api/auth/login/route.ts

import { NextResponse } from 'next/server';
import bcrypt from 'bcryptjs';
import crypto from 'crypto';
import { Redis } from '@upstash/redis';
import prismadb from '@/lib/prismadb';

const redis = Redis.fromEnv();

export async function POST(req: Request) {
  try {
    const { email, password } = await req.json();
    
    // 1. Find user and verify password
    const user = await prismadb.user.findUnique({
      where: { email },
      include: { subscription: true }
    });
    
    if (!user) {
      return NextResponse.json({ error: 'Invalid credentials' }, { status: 401 });
    }
    
    const validPassword = await bcrypt.compare(password, user.passwordHash);
    if (!validPassword) {
      return NextResponse.json({ error: 'Invalid credentials' }, { status: 401 });
    }
    
    // 2. Generate session ID
    const sessionId = crypto.randomUUID();
    
    // 3. Store session data in Redis
    const sessionData = {
      userId: user.id,
      email: user.email,
      name: user.name,
      isActive: user.isActive,
      subscriptionStatus: user.subscription?.status || 'NONE',
      subscriptionExpiry: user.subscription?.currentPeriodEnd,
      gracePeriod: user.gracePeriod,
      createdAt: new Date().toISOString()
    };
    
    // ⭐ Store in Redis with 7-day expiry
    await redis.setex(
      `session:${sessionId}`,
      7 * 24 * 60 * 60, // 7 days in seconds
      JSON.stringify(sessionData)
    );
    
    // 4. Set session ID in cookie
    const response = NextResponse.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        name: user.name
      }
    });
    
    response.cookies.set('sessionId', sessionId, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'lax',
      maxAge: 7 * 24 * 60 * 60, // 7 days
      path: '/'
    });
    
    return response;
    
  } catch (error) {
    console.error('[LOGIN_ERROR]', error);
    return NextResponse.json({ error: 'Internal error' }, { status: 500 });
  }
}
```

**Middleware (Edge Runtime with Redis):**

```typescript
// File: src/middleware.ts

import { NextRequest, NextResponse } from 'next/server';
import { Redis } from '@upstash/redis'; // ⭐ Edge-compatible!

const redis = Redis.fromEnv();

export async function middleware(req: NextRequest) {
  // 1. Extract session ID from cookie
  const sessionId = req.cookies.get('sessionId')?.value;
  
  if (!sessionId) {
    return NextResponse.redirect(new URL('/login', req.url));
  }
  
  try {
    // 2. ⭐ Query Redis (works in edge runtime!)
    const sessionData = await redis.get(`session:${sessionId}`);
    
    if (!sessionData) {
      // Session expired or invalid
      return NextResponse.redirect(new URL('/login', req.url));
    }
    
    // 3. Parse session data
    const session = typeof sessionData === 'string' 
      ? JSON.parse(sessionData) 
      : sessionData;
    
    // 4. ⭐ Check subscription from Redis (NO PostgreSQL query!)
    if (!session.isActive || session.subscriptionStatus === 'EXPIRED') {
      return NextResponse.json(
        {
          error: 'Subscription expired',
          code: 'SUBSCRIPTION_EXPIRED',
          status: session.subscriptionStatus
        },
        { 
          status: 402,
          headers: {
            'X-Subscription-Status': session.subscriptionStatus
          }
        }
      );
    }
    
    // 5. Add user info to headers
    const response = NextResponse.next();
    response.headers.set('X-User-Id', session.userId);
    response.headers.set('X-Email', session.email);
    response.headers.set('X-Subscription-Status', session.subscriptionStatus);
    
    if (session.gracePeriod) {
      response.headers.set('X-Grace-Period', 'true');
    }
    
    return response;
    
  } catch (error) {
    console.error('[MIDDLEWARE_SESSION_ERROR]', error);
    return NextResponse.redirect(new URL('/login', req.url));
  }
}
```

```typescript
// File: src/app/api/webhook/razorpay/subscription/route.ts

import { Redis } from '@upstash/redis';

const redis = Redis.fromEnv();

async function handleSubscriptionExpired(subscription: any) {
  // 1. Update PostgreSQL
  await prismadb.subscription.update({
    where: { id: subscription.id },
    data: { status: 'EXPIRED' }
  });
  
  await prismadb.user.update({
    where: { id: subscription.userId },
    data: { isActive: false }
  });
  
  // 2. ⭐ Update ALL active sessions in Redis (instant effect!)
  const sessionKeys = await redis.keys(`session:*`);
  
  for (const key of sessionKeys) {
    const sessionData = await redis.get(key);
    
    if (sessionData && sessionData.userId === subscription.userId) {
      // Update session with new subscription status
      await redis.set(
        key,
        JSON.stringify({
          ...sessionData,
          isActive: false,
          subscriptionStatus: 'EXPIRED'
        }),
        {
          ex: 7 * 24 * 60 * 60 // Keep same expiry
        }
      );
    }
  }
  
  console.log(`✅ Updated sessions for user ${subscription.userId}`);
}
```
