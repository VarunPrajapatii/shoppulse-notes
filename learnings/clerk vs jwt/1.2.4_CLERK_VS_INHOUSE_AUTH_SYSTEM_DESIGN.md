### **Phase 1 → Phase 2: Adding Redis Cache**

```typescript
const migrationPhase1To2 = {
  currentState: {
    auth: "Clerk",
    subscriptionCheck: "In-memory cache (handler)",
    performance: "~15ms (99% cached)",
    limitations: [
      "Cache per server (not shared)",
      "Can't block at edge",
      "Subscription changes take up to 5 min to reflect"
    ]
  },
  
  targetState: {
    auth: "Clerk (unchanged)",
    subscriptionCheck: "Redis (shared cache)",
    performance: "~15ms (edge blocking)",
    improvements: [
      "Shared cache (consistent across servers)",
      "Edge-level blocking (no origin hit)",
      "Instant invalidation (webhook → Redis → effect)"
    ]
  },
  
  migrationSteps: [
    {
      step: "1. Set up Redis (Upstash)",
      duration: "30 minutes",
      
      actions: [
        "Sign up: https://upstash.com",
        "Create database (free tier)",
        "Copy UPSTASH_REDIS_REST_URL",
        "Copy UPSTASH_REDIS_REST_TOKEN",
        "Add to .env.local (local testing)",
        "Add to Vercel environment variables (production)"
      ],
      
      verification: `
        // Test connection
        import { Redis } from '@upstash/redis';
        
        const redis = Redis.fromEnv();
        
        await redis.set('test', 'hello');
        const value = await redis.get('test');
        console.log(value);  // Should print: hello
      `
    },
    
    {
      step: "2. Create Redis helper functions",
      duration: "1 hour",
      
      file: "src/lib/redis.ts",
      
      code: `
        import { Redis } from '@upstash/redis';
        import prismadb from '@/lib/prismadb';
        
        export const redis = new Redis({
          url: process.env.UPSTASH_REDIS_REST_URL!,
          token: process.env.UPSTASH_REDIS_REST_TOKEN!
        });
        
        export async function cacheUserSubscription(userId: string) {
          const user = await prismadb.user.findUnique({
            where: { id: userId },
            select: {
              isActive: true,
              gracePeriod: true,
              subscription: {
                select: {
                  status: true,
                  currentPeriodEnd: true,
                  planId: true
                }
              }
            }
          });
          
          if (!user) return null;
          
          const cacheData = {
            isActive: user.isActive,
            gracePeriod: user.gracePeriod,
            subscriptionStatus: user.subscription?.status || 'NONE',
            subscriptionExpiry: user.subscription?.currentPeriodEnd?.toISOString(),
            planId: user.subscription?.planId,
            cachedAt: new Date().toISOString()
          };
          
          await redis.setex(
            \`user:\${userId}:subscription\`,
            300,  // 5 minutes
            JSON.stringify(cacheData)
          );
          
          return cacheData;
        }
        
        export async function getCachedSubscription(userId: string) {
          const cached = await redis.get(\`user:\${userId}:subscription\`);
          
          if (!cached) return null;
          
          return typeof cached === 'string' ? JSON.parse(cached) : cached;
        }
        
        export async function invalidateSubscriptionCache(userId: string) {
          await redis.del(\`user:\${userId}:subscription\`);
        }
      `
    },
    
    {
      step: "3. Update middleware to check Redis",
      duration: "1 hour",
      
      file: "src/middleware.ts",
      
      before: `
        export default clerkMiddleware(async (auth, req) => {
          const { userId } = await auth();
          
          if (!userId) {
            return NextResponse.redirect('/login');
          }
          
          // ❌ No subscription check (passes to handler)
          return NextResponse.next();
        });
      `,
      
      after: `
        import { getCachedSubscription, cacheUserSubscription } from '@/lib/redis';
        
        export default clerkMiddleware(async (auth, req) => {
          const { userId } = await auth();
          
          if (!userId) {
            return NextResponse.redirect('/login');
          }
          
          // ✅ Check subscription from Redis
          let subscription = await getCachedSubscription(userId);
          
          if (!subscription) {
            // Cache miss: fetch and cache
            subscription = await cacheUserSubscription(userId);
          }
          
          if (!subscription) {
            return NextResponse.json({ error: 'User not found' }, { status: 404 });
          }
          
          // ✅ Block at edge if expired
          const allowedStatuses = ['ACTIVE', 'COMPLETING', 'PAST_DUE'];
          if (!subscription.isActive || !allowedStatuses.includes(subscription.subscriptionStatus)) {
            return NextResponse.json(
              {
                error: 'Subscription expired',
                code: 'SUBSCRIPTION_EXPIRED',
                status: subscription.subscriptionStatus
              },
              { status: 402 }
            );
          }
          
          // Add to headers
          const response = NextResponse.next();
          response.headers.set('X-User-Id', userId);
          response.headers.set('X-Subscription-Status', subscription.subscriptionStatus);
          
          if (subscription.gracePeriod) {
            response.headers.set('X-Grace-Period', 'true');
          }
          
          return response;
        });
      `
    },
    
    {
      step: "4. Update webhooks to invalidate Redis",
      duration: "30 minutes",
      
      file: "src/app/api/webhook/razorpay/subscription/route.ts",
      
      changes: `
        import { invalidateSubscriptionCache } from '@/lib/redis';
        
        async function handleSubscriptionExpired(subscription: any) {
          // 1. Update PostgreSQL
          await prismadb.subscription.update({
            where: { id: subscription.id },
            data: { status: 'EXPIRED' }
          });
          
          await prismadb.user.update({
            where: { id: subscription.userId },
            data: { isActive: false }
          });
          
          // 2. ✅ Invalidate Redis cache (NEW)
          await invalidateSubscriptionCache(subscription.userId);
          
          console.log(\`✅ Invalidated cache for user \${subscription.userId}\`);
        }
        
        async function handleSubscriptionActivated(subscription: any) {
          // Update database...
          
          // ✅ Invalidate cache (NEW)
          await invalidateSubscriptionCache(subscription.userId);
        }
        
        // Apply to all webhook handlers:
        // - subscription.halted
        // - subscription.activated
        // - subscription.completed
        // - subscription.paused
      `
    },
    
    {
      step: "5. Simplify API handlers (remove withStoreAccessCheck)",
      duration: "1 hour",
      
      rationale: "Subscription now checked in middleware, handlers don't need to re-check",
      
      before: `
        export async function GET(req: Request) {
          return withStoreAccessCheck(req, async (userId, store) => {
            // withStoreAccessCheck queries DB for subscription ❌
            
            const products = await prismadb.product.findMany({
              where: { storeId: store.id }
            });
            
            return NextResponse.json(products);
          });
        }
      `,
      
      after: `
        export async function GET(req: Request, { params }: any) {
          // ✅ Already checked in middleware
          const userId = req.headers.get('x-user-id')!;
          const { storeId } = await params;
          
          // Just verify store ownership
          const store = await prismadb.store.findFirst({
            where: { id: storeId, userId }
          });
          
          if (!store) {
            return NextResponse.json({ error: 'Store not found' }, { status: 404 });
          }
          
          // Fetch products (no subscription check)
          const products = await prismadb.product.findMany({
            where: { storeId }
          });
          
          return NextResponse.json(products);
        }
      `
    },
    
    {
      step: "6. Test locally",
      duration: "1 hour",
      
      tests: [
        {
          test: "Cache hit",
          steps: [
            "1. Make request → logs 'Cache miss, fetching from DB'",
            "2. Make same request → logs 'Cache hit'",
            "3. Verify: Second request faster (<20ms)"
          ]
        },
        {
          test: "Webhook invalidation",
          steps: [
            "1. Make request → cached",
            "2. Trigger webhook (simulate subscription.halted)",
            "3. Make request → cache miss, fetches fresh data",
            "4. Verify: User blocked (402 error)"
          ]
        },
        {
          test: "Cache expiry",
          steps: [
            "1. Make request → cached",
            "2. Wait 5 minutes",
            "3. Make request → cache miss (TTL expired)",
            "4. Verify: Fresh data fetched"
          ]
        }
      ],
      
      testScript: `
```

### **Phase 2 → Phase 3: Migrating to In-House JWT**

```typescript
const migrationPhase2To3 = {
  when: [
    "100K+ MAU (Clerk cost >$1,500/mo)",
    "Need full control over auth",
    "Custom OAuth integrations",
    "Edge performance critical"

            // src/lib/jwt.ts
            import jwt from 'jsonwebtoken';
            
            const ACCESS_TOKEN_SECRET = process.env.JWT_SECRET!;
            const REFRESH_TOKEN_SECRET = process.env.JWT_REFRESH_SECRET!;
            
            export function generateAccessToken(payload: any) {
              return jwt.sign(
                payload,
                ACCESS_TOKEN_SECRET,
                { expiresIn: '15m' }
              );
            }
            
            export function generateRefreshToken(payload: any) {
              return jwt.sign(
                { userId: payload.userId },
                REFRESH_TOKEN_SECRET,
                { expiresIn: '7d' }
              );
            }
            
            export function verifyAccessToken(token: string) {
              return jwt.verify(token, ACCESS_TOKEN_SECRET);
            }
            
            export function verifyRefreshToken(token: string) {
              return jwt.verify(token, REFRESH_TOKEN_SECRET);
            }
          `
        },
        
        {
          task: "2. Build login endpoint",
          file: "src/app/api/auth/login/route.ts",

            "Email/password authentication",

            "Cookie setting (httpOnly, secure)",
            "Fetch subscription on login"
          ]

        {
          task: "3. Build refresh endpoint",
          file: "src/app/api/auth/refresh/route.ts",

            1. Client sends refresh token (cookie)
            2. Verify refresh token
            3. Query DB for latest user data (subscription)
            4. Generate new access token with updated claims
            5. Return new access token
            6. Optionally: Rotate refresh token
          `
        },
        
        {
          task: "4. Build logout endpoint",
          file: "src/app/api/auth/logout/route.ts",

          ]

          task: "5. Update middleware for JWT",

            import { jwtVerify } from 'jose';
            
            export async function middleware(req: NextRequest) {
              const token = req.cookies.get('accessToken')?.value;
              
              if (!token) {
                return NextResponse.redirect('/login');
              }
              
              const secret = new TextEncoder().encode(process.env.JWT_SECRET);
              const { payload } = await jwtVerify(token, secret);
              
              // ✅ Check subscription from JWT claims
              if (payload.subscriptionStatus === 'EXPIRED') {
                return NextResponse.json({ error: 'Expired' }, { status: 402 });
              }
              
              return NextResponse.next();
            }
          `
        },
        
        {
          task: "6. Build OAuth integrations",
          files: [
            "src/app/api/auth/oauth/google/route.ts",

      ]
    },
    
    {
      phase: "C. Migrate users from Clerk",
      duration: "3 days",
      
      steps: [

          step: "1. Export users from Clerk",

            "user_id",
            "email",
            "first_name",
            "last_name",
            "created_at",
            "email_verified"
          ]
        },

                // New auth (JWT)

              }

              if (clerkSession) {
                // Old auth (Clerk)

              }

    {
```

## 🎤 Interview Q&A Scenarios

### **Scenario 1: Auth System Design**

```typescript
const interviewScenario1 = {
  question: `
    You're building a SaaS platform with subscription-based access.
    Walk me through your authentication and authorization architecture.
  `,
  
  answer: {
    // Start with high-level overview
    step1_overview: `
      "I'd design a two-layer system:
      
      1. Authentication Layer: Verify who the user is
         - Options: Clerk (third-party), JWT (in-house), or sessions
         - Handles: Login, OAuth, session management
      
      2. Authorization Layer: Verify what they can access
         - Check subscription status
         - Enforce feature flags
         - Apply rate limits
      
      For a startup, I'd recommend Clerk (phase 1) with plan to migrate
      to JWT later (phase 2) if scale demands."
    `,
    
    // Deep dive when asked
    step2_deepdive: `
      "Let me walk through the request flow:
      
      1. User logs in → Clerk generates session (OAuth + email/password)
      
      2. Session stored in httpOnly cookie (XSS protection)
      
      3. Every request:
         Edge Middleware (Vercel Global):
           ├─ Verify session with Clerk (~5ms)
           ├─ Extract userId
           ├─ Check subscription from Redis cache (~5ms)
           ├─ If expired → Block with 402 (Payment Required)
           └─ If active → Pass to origin server
         
         Origin Server (Node.js):
           ├─ Process business logic
           ├─ Query database (Prisma)
           └─ Return response
      
      4. Total latency: ~20ms (edge) + 50ms (server) = 70ms
      
      Key design decisions:
      - Redis cache: 5-min TTL, 99% hit rate, reduces DB load 100x
      - Edge blocking: Expired users blocked before hitting server
      - Webhook invalidation: Payment fails → Redis cleared → instant block"
    `,
    
    // Follow-up: Why Redis?
    step3_whyRedis: `
      "Redis solves three problems:
      
      1. Shared cache: In-memory cache is per-server, not consistent
         Redis: All servers see same data
      
      2. Edge compatibility: PostgreSQL doesn't work in edge runtime
         Redis HTTP API (Upstash): Works at edge, 5ms latency
      
      3. Instant invalidation: Webhook → Redis → next request sees change
         In-memory cache: Takes up to 5 minutes (TTL)
      
      Trade-off: Adds cost ($20/mo) and complexity, but worth it at scale."
    `,
    
    // Follow-up: Why not JWT?
    step4_whyNotJWT: `
      "JWT would be even faster (~2ms) since subscription is in token claims.
      
      But there's a trade-off:
      
      JWT Pros:
      ✅ Stateless (no Redis query)
      ✅ Fast (<2ms verification)
      ✅ Edge-compatible
      
      JWT Cons:
      ❌ Can't invalidate before expiry (user stays active for 15 min after payment fails)
      ❌ Must build auth yourself (OAuth, user management, password reset)
      ❌ 3-4 weeks development time
      
      With Clerk, I get OAuth and user management for free, worth the 5ms latency.
      
      I'd migrate to JWT at 100K MAU when Clerk costs $1,800/mo."
    `
  }
};
```

```typescript
const interviewScenario2 = {

      3. Database connections: 500K RPS needs 5000 connections 🔴
      4. Edge middleware: 500K RPS × 10ms = 5000 CPU cores 🟡
      
      Solutions:
      1. Migrate to JWT (eliminate Clerk)
      2. Redis cluster (sharding by userId)
      3. Database read replicas (auth queries → replica)
      4. Edge compute auto-scales (Vercel handles)"
    `,
    
    step2_jwtMigration: `
      "JWT Migration (Phase 1):
      
      Why: $20K/mo Clerk → $0, 10ms → 2ms latency

      1. Build JWT auth (3-4 weeks)

      3. Middleware verifies JWT (no Redis query for auth)
      4. Refresh token every 15 min (fetch latest subscription)
      
      Impact:
      - Cost: $20,000/mo → $0
      - Latency: 10ms → 2ms (5x faster)
      - Complexity: High (maintain auth yourself)
      
      Trade-off worth it at this scale."
    `,
    
    step3_redisCluster: `
      "Redis Cluster (Phase 2):
      
      Why: 500K RPS > single Redis capacity (~100K ops/sec)

      1. Redis Cluster with 5 nodes (sharding)

      3. Each node: 100K ops/sec = 500K total

      Cost:
      - Upstash Enterprise: ~$1,000/mo
      - AWS ElastiCache (5× r6g.xlarge): ~$800/mo
      
      Benefit:
      - Linear scaling (add nodes for more capacity)
```
