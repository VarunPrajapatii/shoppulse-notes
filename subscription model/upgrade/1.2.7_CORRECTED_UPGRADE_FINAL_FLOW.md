## Overview

**New Approach (OFFICIAL)**:

- Use `razorpay.subscriptions.update()` API
- Razorpay calculates proration automatically
- No checkout needed (uses saved payment method)
- Handle `subscription.updated` webhook
- Simple logic with same subscription ID

## Critical Rules (Must Know!)

### Rule 1: Minimum Charge Difference
⚠️ **The prorated charge difference must be at least ₹0.50**

If the difference between new plan cost and remaining credit is less than ₹0.50, the update will fail.

```typescript
// Example: Too small difference
Current: ₹100/month, 29 days remaining = ₹96.67 credit
New: ₹100.20/month
Difference: ₹100.20 - ₹96.67 = ₹3.53 ❌ Might round to < ₹0.50
```

### Rule 2: Update Creates Invoice (Not Regular Charge)
- Razorpay creates an **invoice** for the difference amount
- Charges the invoice automatically
- If charge succeeds → `subscription.updated` webhook fires
- If charge fails → Subscription stays on old plan (automatic rollback)

### Rule 3: No subscription.charged Webhook
⚠️ **Only `subscription.updated` fires for immediate upgrades**

Unlike regular billing cycles, upgrade charges don't trigger `subscription.charged`. The payment happens as part of the invoice created during the update process.

## Flow Architecture (Official Method)

### Complete User Journey

```
User clicks "Upgrade"
  → Preview API (calculate proration + show details) [FOR DISPLAY ONLY]
  → User confirms in dialog
  → Execute API (call razorpay.subscriptions.update())
  → Razorpay creates invoice & charges automatically
  → If successful: subscription.updated webhook fires
  → If failed: No webhook, subscription stays on old plan
  → Database updated with new plan (only after webhook)
  → User sees success message
  → Dashboard shows new plan ✓
```

**Key Differences from Old Flow**:

- ✅ No checkout page
- ✅ No verify endpoint needed
- ✅ No manual cancellation of old subscription
- ✅ Same subscription ID throughout
- ✅ Simpler webhook (50 lines vs 163)

```typescript
    expectedCharge: number      // e.g., ₹15,666.67 (estimated)
    newPeriodStart: string      // ISO date

  reason?: string,
```

```typescript
// src/app/api/subscription/upgrade/preview/route.ts

import { NextResponse } from 'next/server';
import { auth } from '@clerk/nextjs/server';
import prisma from '@/lib/prismadb';
import { differenceInDays, addDays, addMonths } from 'date-fns';

export async function POST(req: Request) {
  try {
    const { userId } = await auth();
    if (!userId) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const { newPlanId } = await req.json();

    // Get user with subscription
    const user = await prisma.user.findUnique({
      where: { id: userId },
      include: {
        subscription: {
          include: { plan: true },
        },
      },
    });

    if (!user?.subscription) {
      return NextResponse.json(
        { error: 'No active subscription found', canUpgrade: false },
        { status: 404 }
      );
    }

    const currentSub = user.subscription;

    // Validate subscription status
    if (!['ACTIVE', 'AUTHENTICATED'].includes(currentSub.status)) {
      return NextResponse.json({
        canUpgrade: false,
        reason: `Subscription must be active to upgrade. Current status: ${currentSub.status}`,
      });
    }

    // Get new plan
    const newPlan = await prisma.subscriptionPlan.findUnique({
      where: { id: newPlanId },
    });

    if (!newPlan) {
      return NextResponse.json(
        { error: 'Plan not found', canUpgrade: false },
        { status: 404 }
      );
    }

    // Validate it's an upgrade (higher price)
    const currentPrice = Number(currentSub.plan.price);
    const newPrice = Number(newPlan.price);

    if (newPrice <= currentPrice) {
      return NextResponse.json({
        canUpgrade: false,
        reason:
          'New plan must be more expensive. Use downgrade endpoint for cheaper plans.',
      });
    }

    // ⚠️ CRITICAL: Validate minimum charge difference (₹0.50)
    const credit = currentPrice * (daysRemaining / totalDays);
    const expectedCharge = Math.max(0, newPrice - credit);
    
    if (expectedCharge > 0 && expectedCharge < 0.50) {
      return NextResponse.json({
        canUpgrade: false,
        reason:
          'The upgrade amount is too small (less than ₹0.50). Please wait a few more days or choose a different plan.',
      });
    }

    // Calculate proration (FOR DISPLAY ONLY)
    const now = new Date();
    const periodEnd = new Date(currentSub.currentPeriodEnd);
    const daysRemaining = Math.max(0, differenceInDays(periodEnd, now));

    const totalDays =
      currentSub.billingCycle === 'MONTHLY'
        ? 30
        : currentSub.billingCycle === 'HALFYEARLY'
        ? 180
        : 365;

    const credit = currentPrice * (daysRemaining / totalDays);
    const expectedCharge = Math.max(0, newPrice - credit);

    // Calculate new period
    const newPeriodStart = now;
    const newPeriodEnd =
      newPlan.billingCycle === 'MONTHLY'
        ? addMonths(now, 1)
        : newPlan.billingCycle === 'HALFYEARLY'
        ? addMonths(now, 6)
        : addMonths(now, 12);

    return NextResponse.json({
      canUpgrade: true,
      currentPlan: {
        id: currentSub.plan.id,
        name: currentSub.plan.name,
        displayName: currentSub.plan.displayName,
        price: currentPrice,
        billingCycle: currentSub.plan.billingCycle,
      },
      newPlan: {
        id: newPlan.id,
        name: newPlan.name,
        displayName: newPlan.displayName,
        price: newPrice,
        billingCycle: newPlan.billingCycle,
        maxProducts: newPlan.maxProducts,
        maxOrders: newPlan.maxOrders,
        maxCategories: newPlan.maxCategories,
        maxBillboards: newPlan.maxBillboards,
        maxStorageGB: newPlan.maxStorageGB,
      },
      proration: {
        daysRemaining,
        totalDaysInCycle: totalDays,
        credit: Math.round(credit * 100) / 100,
        expectedCharge: Math.round(expectedCharge * 100) / 100,
        newPeriodStart: newPeriodStart.toISOString(),
        newPeriodEnd: newPeriodEnd.toISOString(),
      },
      note: 'These are estimates. Razorpay will calculate the actual proration.',
    });
  } catch (error) {
    console.error('[UPGRADE_PREVIEW_ERROR]', error);
    return NextResponse.json(
      { error: 'Failed to preview upgrade', canUpgrade: false },
      { status: 500 }
    );
  }
}
```

```typescript
// src/app/api/subscription/upgrade/execute/route.ts


import { NextResponse } from 'next/server';
import { auth } from '@clerk/nextjs/server';
import prisma from '@/lib/prismadb';
import razorpay from '@/lib/razorpay';

export async function POST(req: Request) {
  try {
    const { userId } = await auth();
    if (!userId) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const { newPlanId } = await req.json();

    // Get user with subscription
    const user = await prisma.user.findUnique({
      where: { id: userId },
      include: {
        subscription: {
          include: { plan: true },
        },
      },
    });

    if (!user?.subscription) {
      return NextResponse.json(
        { error: 'No active subscription' },
        { status: 404 }
      );
    }

    const currentSub = user.subscription;

    // Validate status
    if (!['ACTIVE', 'AUTHENTICATED'].includes(currentSub.status)) {
      return NextResponse.json(
        {
          error: 'Subscription must be active to upgrade',
          currentStatus: currentSub.status,
        },
        { status: 400 }
      );
    }

    // Get new plan
    const newPlan = await prisma.subscriptionPlan.findUnique({
      where: { id: newPlanId },
    });

    if (!newPlan) {
      return NextResponse.json({ error: 'Plan not found' }, { status: 404 });
    }

    // Validate upgrade
    if (Number(newPlan.price) <= Number(currentSub.plan.price)) {
      return NextResponse.json(
        { error: 'New plan must be more expensive' },
        { status: 400 }
      );
    }

    console.log(
      `[UPGRADE] Updating subscription ${currentSub.razorpaySubscriptionId}`
    );
    console.log(
      `[UPGRADE] Old plan: ${currentSub.plan.name} (${currentSub.plan.razorpayPlanId})`
    );
    console.log(
      `[UPGRADE] New plan: ${newPlan.name} (${newPlan.razorpayPlanId})`
    );

    // ✅ CALL OFFICIAL UPDATE API
    const updatedSubscription = await razorpay.subscriptions.update(
      currentSub.razorpaySubscriptionId!,
      {
        plan_id: newPlan.razorpayPlanId!,
        schedule_change_at: 'now',
        customer_notify: true,
      }
    );

    console.log('[UPGRADE] Razorpay subscription updated successfully');
    console.log(
      `[UPGRADE] Has scheduled changes: ${updatedSubscription.has_scheduled_changes}`
    );

    // Update database with metadata (webhook will complete the update)
    await prisma.subscription.update({
      where: { id: currentSub.id },
      data: {
        hasScheduledChanges: updatedSubscription.has_scheduled_changes || false,
        scheduledPlanId: null, // note: for upgrade i always want to apply immediately
        changeScheduledAt: updatedSubscription.change_scheduled_at
          ? new Date(updatedSubscription.change_scheduled_at * 1000)
          : null,
        metadata: {
          ...((currentSub.metadata as any) || {}),
          lastUpgradeAttempt: new Date().toISOString(),
          upgradeMethod: 'UPDATE_API',
        },
      },
    });

    // Create history record
    await prisma.subscriptionHistory.create({
      data: {
        subscriptionId: currentSub.id,
        action: 'UPGRADED',
        fromPlanId: currentSub.planId,
        toPlanId: newPlanId,
        fromStatus: currentSub.status,
        toStatus: currentSub.status, // Status stays same, plan changes
        reason: 'Immediate upgrade via UPDATE API',
        metadata: {
          method: 'UPDATE_API',
          razorpaySubscriptionId: updatedSubscription.id,
          hasScheduledChanges: false,
        },
      },
    });

    return NextResponse.json({
      success: true,
      message: 'Upgrade processed successfully! Your plan has been updated and charged.',
      subscription: {
        id: updatedSubscription.id,
        planId: newPlan.razorpayPlanId,
        status: updatedSubscription.status,
      },
    });
  } catch (error: any) {
    console.error('[UPGRADE_EXECUTE_ERROR]', error);

    // Handle Razorpay errors
    if (error.error?.code === 'BAD_REQUEST_ERROR') {
      return NextResponse.json(
        {
          error: error.error.description || 'Invalid request to Razorpay',
          code: error.error.code,
        },
        { status: 400 }
      );
    }

    return NextResponse.json(
      { error: 'Failed to upgrade subscription' },
      { status: 500 }
    );
  }
}
```

## Phase 3: Webhook Handling (subscription.updated)

### ⚠️ CRITICAL: Only subscription.updated Fires

**Important**: For immediate upgrades using UPDATE API:
- ✅ **subscription.updated** webhook fires (if payment succeeds)
- ❌ **subscription.charged** webhook does NOT fire
- The payment happens via an invoice created during update
- This is different from regular billing cycle charges

### Handle subscription.updated Event

**File**: `src/app/api/webhook/razorpay/subscription/route.ts`

**Add new handler**:

```typescript
async function handleSubscriptionUpdated(subscriptionEntity: any) {
  console.log('🔄 [WEBHOOK] subscription.updated received');
  console.log('[WEBHOOK] Subscription ID:', subscriptionEntity.id);

  const razorpaySubId = subscriptionEntity.id;

  // Find subscription in database
  const dbSub = await prisma.subscription.findUnique({
    where: { razorpaySubscriptionId: razorpaySubId },
    include: { plan: true, user: true },
  });

  if (!dbSub) {
    console.error(
      '[WEBHOOK] Subscription not found in database:',
      razorpaySubId
    );
    return;
  }

  // Get new plan from Razorpay response
  const newRazorpayPlanId = subscriptionEntity.plan_id;

  console.log('[WEBHOOK] Current DB plan:', dbSub.plan.razorpayPlanId);
  console.log('[WEBHOOK] New Razorpay plan:', newRazorpayPlanId);

  // ⚠️ IMPORTANT: This webhook only fires if payment succeeded
  // If upgrade payment failed, this webhook won't be triggered
  // Razorpay handles rollback automatically

  // Find new plan in database
  const newPlan = await prisma.subscriptionPlan.findUnique({
    where: { razorpayPlanId: newRazorpayPlanId },
  });

  if (!newPlan) {
    console.error(
      '[WEBHOOK] New plan not found in database:',
      newRazorpayPlanId
    );
    return;
  }

  // Check if plan actually changed
  if (dbSub.planId === newPlan.id) {
    console.log('[WEBHOOK] Plan unchanged, skipping update');
    return;
  }

  console.log(`[WEBHOOK] Plan changed: ${dbSub.plan.name} → ${newPlan.name}`);

  // Update subscription
  await prisma.subscription.update({
    where: { id: dbSub.id },
    data: {
      planId: newPlan.id,
      razorpayPlanId: newRazorpayPlanId,
      billingCycle: newPlan.billingCycle,
      status: 'ACTIVE',
      currentPeriodStart: new Date(subscriptionEntity.current_start * 1000),
      currentPeriodEnd: new Date(subscriptionEntity.current_end * 1000),
      paidCount: subscriptionEntity.paid_count || 0,
      remainingCount: subscriptionEntity.remaining_count,
      hasScheduledChanges: subscriptionEntity.has_scheduled_changes || false,
      scheduledPlanId: null,
      changeScheduledAt: null,
      // NOTE: if user cancled and then upgraded then, (but RP cancellations are immutable) Clear cancellation if was cancelled
      canceledAt: null,
      cancelAtPeriodEnd: false,
      autoRenew: true,
      metadata: {
        ...((dbSub.metadata as any) || {}),
        lastWebhookUpdate: new Date().toISOString(),
        webhookEvent: 'subscription.updated',
      },
    },
  });

  // Create history record
  await prisma.subscriptionHistory.create({
    data: {
      subscriptionId: dbSub.id,
      action: 'PLAN_CHANGED',
      fromPlanId: dbSub.planId,
      toPlanId: newPlan.id,
      fromStatus: dbSub.status,
      toStatus: 'ACTIVE',
      reason: 'Plan changed via subscription.updated webhook',
      metadata: {
        source: 'webhook',
        event: 'subscription.updated',
        razorpaySubscriptionId: razorpaySubId,
        paidCount: subscriptionEntity.paid_count,
        oldPlan: dbSub.plan.name,
        newPlan: newPlan.name,
      },
    },
  });

  // Invalidate store cache
  if (dbSub.user.storeId) {
    await invalidateStoreCache(dbSub.user.storeId);
  }

  console.log(
    `✅ [WEBHOOK] Subscription updated successfully: ${dbSub.plan.name} → ${newPlan.name}`
  );
}
```
