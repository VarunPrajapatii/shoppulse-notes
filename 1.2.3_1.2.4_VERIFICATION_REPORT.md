```typescript
export function calculateProrationCredit(
  planPrice: number,
  billingCycle: BillingCycle,
  currentPeriodEnd: Date
): {
  credit: number;
  daysRemaining: number;
  hoursRemaining: number;
  percentageUnused: number;
} {
  const hoursRemaining = differenceInHours(currentPeriodEnd, now);
  const fractionalDaysRemaining = hoursRemaining / 24;
  const totalDays = getBillingCycleDays(billingCycle);
  const credit = planPrice * (fractionalDaysRemaining / totalDays);
  
  return {
    credit: Math.round(credit * 100) / 100, // Round to 2 decimals
    // ...
  };
}
```

```typescript
      action: 'DELETE',

  return {
    canDowngrade: overages.length === 0,
    reason: overages.length > 0 ? 'Usage exceeds new plan limits' : undefined,

  };
```

```typescript
// GET /api/cron/apply-scheduled-plan-changes
export async function GET(req: Request) {
  // Verify cron secret
  const authHeader = req.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  
  const now = new Date();
  
  // Find subscriptions with scheduled changes where period has ended
  const subscriptionsWithChanges = await prisma.subscription.findMany({
    where: {
      status: 'ACTIVE',
      currentPeriodEnd: { lte: now },
      metadata: {
        path: ['hasScheduledChange'],
        equals: true
      }
    },
    include: { plan: true, user: true }
  });
  
  for (const subscription of subscriptionsWithChanges) {
    try {
      const metadata = subscription.metadata as any;
      const scheduledChange = metadata.scheduledChange;
      
      // Get new plan
      const newPlan = await prisma.subscriptionPlan.findUnique({
        where: { id: scheduledChange.newPlanId }
      });
      
      // Cancel old Razorpay subscription
      await razorpay.subscriptions.cancel(
        subscription.razorpaySubscriptionId!,
        false
      );
      
      // Create new Razorpay subscription
      const newRazorpaySub = await razorpay.subscriptions.create({
        plan_id: newPlan.razorpayPlanId!,
        customer_id: subscription.razorpayCustomerId,
        total_count: getBillingCycleCount(newPlan.billingCycle),
        start_at: Math.floor(Date.now() / 1000)
      });
      
      // Update database
      await prisma.subscription.update({
        where: { id: subscription.id },
        data: {
          planId: newPlan.id,
          razorpaySubscriptionId: newRazorpaySub.id,
          razorpayPlanId: newPlan.razorpayPlanId,
          currentPeriodStart: new Date(),
          currentPeriodEnd: calculateNewPeriodEnd(new Date(), newPlan.billingCycle),
          metadata: {
            hasScheduledChange: false,
            scheduledChange: null
          }
        }
      });
      
      // Create history
      await prisma.subscriptionHistory.create({
        data: {
          subscriptionId: subscription.id,
          action: 'DOWNGRADED',
          toPlanId: newPlan.id,
          reason: 'Scheduled downgrade applied at cycle end'
        }
      });
      
      console.log(`✅ Applied scheduled downgrade for ${subscription.user.email}`);
    } catch (error) {
      console.error(`❌ Failed to apply scheduled change:`, error);
    }
  }
}
```

```typescript
const retryOperation = async (fn, retries = 3) => {
  for (let i = 0; i < retries; i++) {
    try {
      return await fn();
    } catch (error) {
      if (i === retries - 1) throw error;
      await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
    }
  }
```
