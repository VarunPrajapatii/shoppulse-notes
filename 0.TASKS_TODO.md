## 2️⃣ Add Rate Limiting to API Routes 🟡 HIGH

Priority: HIGH | Effort: 2-3 hours

Why: Prevent abuse of public APIs (products, checkout, webhooks)

#### Implementation:

`npm install @upstash/ratelimit @upstash/redis`

```ts
// lib/rate-limit.ts
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

export const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(100, '10 m'), // 100 requests per 10 minutes
});

// middleware.ts
if (req.nextUrl.pathname.startsWith('/api/')) {
  const ip = req.ip ?? '127.0.0.1';
  const { success } = await ratelimit.limit(ip);

  if (!success) {
    return new NextResponse('Too Many Requests', { status: 429 });
  }
}
```

## 3️⃣ Add Data Recovery Endpoint 🟡 MEDIUM
Priority: MEDIUM | Effort: 1-2 hours

Why: Fix edge case where onboardingComplete=true but no store exists

#### Implementation:

```ts
// app/api/admin/fix-user-state/route.ts
export async function POST(req: Request) {
  const { auth } = await auth();
  
  // Verify admin role
  if (!isAdmin(auth.userId)) {
    return new NextResponse('Forbidden', { status: 403 });
  }

  const { userId } = await req.json();

  const user = await prismadb.user.findUnique({
    where: { id: userId },
    include: { stores: true }
  });

  if (user.onboardingComplete && user.stores.length === 0) {
    // Reset onboarding state
    await prismadb.user.update({
      where: { id: userId },
      data: { onboardingComplete: false, subscriptionId: null }
    });

    return NextResponse.json({ 
      success: true, 
      message: 'User state reset. User can complete onboarding again.' 
    });
  }

  return NextResponse.json({ success: true, message: 'No inconsistency found' });
}
```

Priority: MEDIUM | Effort: 1 hour

Why: Monitor system readiness (trial plan configured, webhooks working)

```ts
// app/api/health/onboarding/route.ts
export async function GET() {
  const trialPlan = await prismadb.subscriptionPlan.findFirst({
    where: { name: 'TRIAL', isActive: true }
  });
  
  return NextResponse.json({
    status: trialPlan ? 'healthy' : 'misconfigured',
    trialPlanConfigured: !!trialPlan,
    timestamp: new Date().toISOString()
  });
}

// app/api/health/webhooks/route.ts
export async function GET() {
  // Check last webhook timestamp
  const lastUser = await prismadb.user.findFirst({
    orderBy: { createdAt: 'desc' }
  });

  const minutesSinceLastWebhook = lastUser
    ? (Date.now() - lastUser.createdAt.getTime()) / 60000
    : null;

  return NextResponse.json({
    status: 'healthy',
    lastWebhookReceived: lastUser?.createdAt,
    minutesSinceLastWebhook,
    timestamp: new Date().toISOString()
  });
}
```

```ts
const resend = new Resend(process.env.RESEND_API_KEY);
```
